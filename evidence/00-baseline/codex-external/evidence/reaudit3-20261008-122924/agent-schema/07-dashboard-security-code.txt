   708	    def do_POST(self):
   709	        if not self._gate():
   710	            return
   711	        path = self.path.split("?", 1)[0]
   712	        if path in ("/api/decide", "/api/v1/decide"):
   713	            self.close_connection = True
   714	            return self._err(405, "WRITE_DISABLED",
   715	                             "第一阶段关闭写接口（decide POST 已禁用）——只读仪表盘", allow="GET, HEAD")
   716	        te = (self.headers.get("Transfer-Encoding") or "").lower()
   717	        if te and te != "identity":
   718	            self.close_connection = True
   719	            return self._err(411, "LENGTH_REQUIRED", "不支持 chunked 请求体")
   720	        try:
   721	            n = int(self.headers.get("Content-Length") or 0)
   722	        except ValueError:
   723	            self.close_connection = True
   724	            return self._err(400, "BAD_LENGTH", "Content-Length 非法")
   725	        if n > MAX_BODY:
   726	            self.close_connection = True
   727	            return self._err(413, "BODY_TOO_LARGE", "请求体超限（>%d 字节）" % MAX_BODY)
   728	        if n > 0:
   729	            ct = (self.headers.get("Content-Type") or "").split(";")[0].strip().lower()
   730	            if ct != "application/json":  # Content-Type 精确校验（仅接受 application/json）
   731	                self.close_connection = True
   732	                return self._err(415, "UNSUPPORTED_MEDIA_TYPE", "Content-Type 必须为 application/json")
   733	            try:
   734	                self.rfile.read(n)
   735	            except OSError:
   736	                return
   737	        self._err(404, "NOT_FOUND", "POST 端点不存在（第一阶段写接口全关）")
