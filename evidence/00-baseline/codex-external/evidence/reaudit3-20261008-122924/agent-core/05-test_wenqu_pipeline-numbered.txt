    61	def t_rule_engine():
    62	    assert SM.stages() == STAGES and len(STAGES) == 7
    63	    assert SM.next_stage("S1_REQUIREMENT") == "S2_SOLUTION"
    64	    assert SM.next_stage("S7_GATE") is None
    65	    expect(StageOrderError, lambda: SM.stage_index("S8_NOPE"))
    66	    expect(StageOrderError, lambda: SM.next_stage("S9"))
    67	    # 前段非 PASSED 不得启动后段（含 CONDITIONAL 也不行）
    68	    for bad in ("PENDING", "RUNNING", "COMPLETED_CONDITIONAL", "FAILED", "WAITING"):
    69	        sts = {"S1_REQUIREMENT": bad, "S2_SOLUTION": "PENDING"}
    70	        expect(StageOrderError, lambda s=bad: SM.ensure_prior_stages_passed(
    71	            {"S1_REQUIREMENT": s}, "S2_SOLUTION"))
    72	    SM.ensure_prior_stages_passed({"S1_REQUIREMENT": "PASSED"}, "S2_SOLUTION")
    73	    # 转换表
    74	    SM.validate_run_transition("CREATED", "RUNNING")
    75	    SM.validate_run_transition("WAITING", "RUNNING")
    76	    SM.validate_run_transition("RUNNING", "SUCCEEDED")
    77	    expect(InvalidTransitionError, lambda: SM.validate_run_transition("CREATED", "WAITING"))
    78	    expect(InvalidTransitionError, lambda: SM.validate_run_transition("SUCCEEDED", "RUNNING"))
    79	    expect(InvalidTransitionError, lambda: SM.validate_run_transition("WAITING", "SUCCEEDED"))
    80	    SM.validate_attempt_transition("PENDING", "RUNNING")
    81	    SM.validate_attempt_transition("RUNNING", "WAITING")
    82	    expect(InvalidTransitionError, lambda: SM.validate_attempt_transition("WAITING", "RUNNING"))
    83	    # 双轴：非 COMPLETED 不得 PASS；PASS 不得带 findings
    84	    SM.validate_outcome("COMPLETED", "PASS")
    85	    expect(InvalidOutcomeError, lambda: SM.validate_outcome("ERROR", "PASS"))
    86	    expect(InvalidOutcomeError, lambda: SM.validate_outcome("TIMEOUT", "PASS"))
    87	    expect(InvalidOutcomeError, lambda: SM.validate_outcome("COMPLETED", "PASS", ["fnd_x"]))
    88	    expect(InvalidOutcomeError, lambda: SM.validate_outcome("COMPLETED", "GREEN"))
    89	    # 九类停等 + 前置规格（含 exm-fuse 的 either-or）
    90	    assert len(list(NineStopTypes)) == 9
    91	    assert set(SM.stop_types()) == {
    92	        "ambiguity", "prod-write", "ddl", "release", "scope",
    93	        "fund-auth", "exm-fuse", "resource", "ext-unavail"}
    94	    SM.validate_preconditions("ambiguity", {"requirement_four_elements": "四要素", "scope_hash": "sh"})
    95	    expect(ApprovalPreconditionError, lambda: SM.validate_preconditions("ambiguity", {"scope_hash": "sh"}))
    96	    expect(ApprovalPreconditionError, lambda: SM.validate_preconditions("ambiguity", None))
    97	    SM.validate_preconditions("exm-fuse", {"adjudication_ref": "EXM-1", "fuse_lifted_evidence": "probe"})
    98	    SM.validate_preconditions("exm-fuse", {"adjudication_ref": "EXM-1", "independent_ruling_ref": "ruling-9"})
    99	    expect(ApprovalPreconditionError, lambda: SM.validate_preconditions("exm-fuse", {"adjudication_ref": "EXM-1"}))
   100	    h = SM.objective_precondition_hash({"a": 1})

   350	def t_concurrent_double_consume():
   351	    tmp = tempfile.mkdtemp(prefix="wenqu-pipe-cc-")
   352	    db = os.path.join(tmp, "events.db")
   353	    store = EventStore(db)
   354	    mgr = RunManager(store)
   355	    run_id = mgr.create_run("task_cc", IDENT)["run_id"]
   356	    mgr.advance_stage(run_id)
   357	    w = mgr.raise_waiting(run_id, "prod-write", "并发双消费")
   358	    v = mgr.get_run(run_id).state_version
   359	    ap, pre = make_resume_approval(run_id, w["stop_event_id"], "prod-write", v,
   360	                                   nonce="nonce-cc-00112233445566")
   361	    # 第二个独立管理器（独立连接，模拟并发进程）
   362	    mgr_b = RunManager(EventStore(db))
   363	    results, errors = [], []
   364	    def worker(m):
   365	        try:
   366	            results.append(m.resume_waiting(run_id, ap, dict(pre)))
   367	        except Exception as e:
   368	            errors.append(e)
   369	    t1 = threading.Thread(target=worker, args=(mgr,))
   370	    t2 = threading.Thread(target=worker, args=(mgr_b,))
   371	    t1.start(); t2.start(); t1.join(); t2.join()
   372	    succ = [r for r in results if r.get("resumed")]
   373	    reuse = [e for e in errors if isinstance(e, ApprovalReuseError)]

   401	def t_conditional_terminal():
   402	    store, mgr = fresh()
   403	    run_id = mgr.create_run("task_cond", IDENT)["run_id"]
   404	    for i in range(6):  # S1..S6 PASS
   405	        mgr.advance_stage(run_id, execution_status="COMPLETED", policy_verdict="PASS")
   406	    out = mgr.advance_stage(run_id, execution_status="COMPLETED", policy_verdict="CONDITIONAL")
   407	    assert out["attempt_status"] == "COMPLETED_CONDITIONAL"
   408	    assert out.get("next_stage") is None
   409	    st = mgr.get_run(run_id)
   410	    assert st.current_stage == "S7_GATE"  # 软终态段仍阻塞推进
   411	    out = mgr.advance_stage(run_id)  # 软终态可重试：attempt 不可变 → 新 attempt
   412	    assert out.get("started_attempt_id"), out
   413	    out = mgr.advance_stage(run_id, execution_status="COMPLETED", policy_verdict="CONDITIONAL")
   414	    # 无 risk 授权 → complete_run fail-closed
   415	    expect(PipelineError, lambda: mgr.complete_run(run_id))
   416	    rec = mgr.authorize_risk(run_id, make_risk_approval(run_id))
   417	    assert rec["consumed"] is True
   418	    # risk 审批不可复用
   419	    expect(ApprovalReuseError, lambda: mgr.authorize_risk(run_id, make_risk_approval(run_id)))
   420	    done = mgr.complete_run(run_id)
   421	    assert done["final_state"] == "COMPLETED_CONDITIONAL"
   422	    assert done["policy_verdict"] == "CONDITIONAL"
   423	    expect(TerminalRunError, lambda: mgr.complete_run(run_id))
   424	    assert store.verify_chain()["ok"]

   471	def t_schema_crosscheck():
   472	    """结构校验器与 approval-v2.schema.json 交叉验证（jsonschema 可用时）。"""
   473	    try:
   474	        import jsonschema
   475	    except ImportError:
   476	        print("  (skip) jsonschema 未安装，跳过交叉验证")
   477	        return
   478	    schema = json.load(open(os.path.join(REPO, "schemas", "approval-v2.schema.json")))
   479	    store, mgr = fresh()
   480	    run_id = mgr.create_run("task_x", IDENT)["run_id"]
   481	    mgr.advance_stage(run_id)
   482	    w = mgr.raise_waiting(run_id, "prod-write", "x")
   483	    v = mgr.get_run(run_id).state_version
   484	    ap, pre = make_resume_approval(run_id, w["stop_event_id"], "prod-write", v)
   485	    # 合法样例：schema 与内置校验器都放行
   486	    jsonschema.validate(ap, schema)
   487	    ApprovalBroker.validate_structure(ap)
   488	    # 篡改样例：两者都拒绝
   489	    for mut in (lambda a: a.update({"schema_version": "1.9"}),
   490	                lambda a: a.update({"decision": "maybe"}),
   491	                lambda a: a.update({"nonce": "short"}),
   492	                lambda a: a.pop("actor"),
   493	                lambda a: a["payload"].pop("objective_precondition_hash")):
   494	        bad = json.loads(json.dumps(ap))
   495	        mut(bad)
   496	        try:
   497	            jsonschema.validate(bad, schema)
   498	            schema_rejects = False
   499	        except jsonschema.ValidationError:
   500	            schema_rejects = True
   501	        try:
   502	            ApprovalBroker.validate_structure(bad)
   503	            mine_rejects = False
   504	        except ApprovalRejected:
   505	            mine_rejects = True
   506	        assert schema_rejects and mine_rejects, f"分歧: schema={schema_rejects} mine={mine_rejects} on {bad}"
