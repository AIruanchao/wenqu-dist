   742	def _manifest_core_digest(manifest: Dict[str, Any]) -> str:
   743	    """manifest 核心 digest：sha256(canonical JSON 去易变字段)。
   744	
   745	    核心 = 除 VOLATILE_MANIFEST_KEYS（build_id/built_at/builder/argv——构建
   746	    身份，同 commit 异时异地重建必变）外的全部字段（schema/commit/version/
   747	    policy_hash/artifact/逐文件清单……）。授权流程在授权构建上取值写入
   748	    roots.manifest_sha256；安装链重建后重算必须等值——方案/文件/commit/
   749	    版本的任何篡改都改变核心，digest 断裂。
   750	    """
   751	    core = {k: v for k, v in manifest.items() if k not in VOLATILE_MANIFEST_KEYS}
   752	    canonical = json.dumps(core, sort_keys=True, separators=(",", ":"),
   753	                           ensure_ascii=False)
   754	    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

   757	def _load_release_roots(roots_path: str) -> Dict[str, Any]:
   758	    """读外部授权 roots 文件（G9-12：安装器只消费，绝不生成）。
   759	
   760	    fail-closed 面（全部 VerifyFail，exit 1）：
   761	    - 文件缺失/不可读/非 JSON 对象；
   762	    - 权限过宽：mode & 0o077 != 0（组/其他**任何**位——含可读——> 0600
   763	      语义即拒：他人可读/可写的文件不是授权根）；
   764	    - schema 不符 / plan_sha256 非 64hex / allowed_commit 空或非列表；
   765	    - 条目 commit 非 40hex / manifest_sha256 非 64hex / commit 重复。

   769	    if not os.path.isfile(roots_path):
   770	        raise VerifyFail(
   771	            f"release-roots 授权文件缺失（fail-closed，无豁免路径）: "
   772	            f"{roots_path!r}——由部署授权流程仓外预置（0600），安装器不生成")
   773	    try:
   774	        st = os.stat(roots_path)
   775	        if st.st_mode & 0o077:
   776	            raise VerifyFail(
   777	                f"release-roots 权限过宽（组/其他任何位，>0600 即拒）: "
   778	                f"{roots_path!r} mode={oct(st.st_mode & 0o7777)}——授权根须"
   779	                f" owner-only（0600），由部署授权流程 chmod 600 预置")
   780	        with open(roots_path, "r", encoding="utf-8") as fh:
   781	            data = json.load(fh)
   782	    except VerifyFail:
   783	        raise
   784	    except (OSError, ValueError) as exc:
   785	        raise VerifyFail(f"release-roots 不可读/非 JSON: {roots_path!r} ({exc})")
   786	    if not isinstance(data, dict) \
   787	            or data.get("schema") != RELEASE_ROOTS_SCHEMA:
   788	        raise VerifyFail(
   789	            f"release-roots schema 不符（期望 {RELEASE_ROOTS_SCHEMA}）: "
   790	            f"{roots_path!r}")
   791	    plan = data.get("plan_sha256")
   792	    if not isinstance(plan, str) or not re.fullmatch(r"[0-9a-f]{64}", plan):
   793	        raise VerifyFail(
   794	            f"release-roots.plan_sha256 非法（应 64 位小写 hex）: {plan!r}")
   795	    allowed = data.get("allowed_commit")
   796	    if not isinstance(allowed, list) or not allowed:
   797	        raise VerifyFail(
   798	            "release-roots.allowed_commit 须为非空列表（外部授权的 commit "
   799	            f"清单）: {type(allowed).__name__}")
   800	    entries: Dict[str, str] = {}
   801	    for i, ent in enumerate(allowed):
   802	        if not isinstance(ent, dict):
   803	            raise VerifyFail(f"allowed_commit[{i}] 非对象: {ent!r}")
   804	        c = ent.get("commit")
   805	        d = ent.get("manifest_sha256")
   806	        if not isinstance(c, str) or not re.fullmatch(r"[0-9a-f]{40}", c):
   807	            raise VerifyFail(
   808	                f"allowed_commit[{i}].commit 非法（应 40 位小写 hex）: {c!r}")
   809	        if not isinstance(d, str) or not re.fullmatch(r"[0-9a-f]{64}", d):
   810	            raise VerifyFail(
   811	                f"allowed_commit[{i}].manifest_sha256 非法（应 64hex）: {d!r}")
   812	        if c in entries:
   813	            raise VerifyFail(f"allowed_commit 重复 commit: {c}")
   814	        entries[c] = d
   815	    return {"plan_sha256": plan, "entries": entries}
