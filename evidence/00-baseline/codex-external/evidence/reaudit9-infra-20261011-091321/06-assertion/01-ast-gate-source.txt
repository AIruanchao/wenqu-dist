   518	    """解析基线：# 注释行跳过；<文件名> <合格数>；非法行/重复条目抛错。"""

   521	        line = raw.strip()

   523	            continue
   524	        parts = line.split()
   525	        if len(parts) != 2 or not parts[1].isdigit():

   527	        if parts[0] in baseline:

   529	        baseline[parts[0]] = int(parts[1])
