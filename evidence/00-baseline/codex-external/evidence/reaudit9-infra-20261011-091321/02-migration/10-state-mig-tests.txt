   283	    每条恰好 inserted=True 一次）、链仍单一线性且 verify_chain 通过。
   284	    附加子场景：20 个 writer 并发追加同一内容 → 恰好 1 行（幂等不双入账）。
   285	    """
   286	    WRITERS, PER_WRITER = 100, 100
   287	    with TmpDir("ac-s3-") as td:
   288	        db = os.path.join(td, "ev.db")

   290	        inserted = [0] * WRITERS
   291	        errors = []

   293	        def writer(w):
   294	            try:
   295	                s = EventStore(db)

   311	            t.join()

   319	                f"inserted 计数不符: {sum(inserted)}"

   321	                store._conn,
   322	                "SELECT COUNT(DISTINCT event_id) FROM pipeline_events")[0][0]
   323	            assert uniq_ids == WRITERS * PER_WRITER, f"重复行: {uniq_ids}"
   324	            verify = store.verify_chain()
   325	            assert verify["ok"] and verify["length"] == WRITERS * PER_WRITER
   326	            assert verify["head"] == head
   327	        finally:
   328	            store.close()
   329	
   330	        # 子场景：同内容并发重放 → 恰好 1 行、1 个 inserted=True、hash 全同
   331	        db2 = os.path.join(td, "idem.db")
   332	        EventStore(db2).close()
   333	        results, ierr = [], []

   335	        def idem_writer():
   336	            try:
   337	                s = EventStore(db2)
   338	                try:
   339	                    results.append(s.append({"same": "retry-payload", "n": 1}))
   340	                finally:
   341	                    s.close()
   342	            except Exception as e:      # noqa: BLE001
   343	                ierr.append(type(e).__name__)
   344	
   345	        threads = [threading.Thread(target=idem_writer) for _ in range(20)]
   346	        for t in threads:
   347	            t.start()
   348	        for t in threads:
   349	            t.join()
   350	        assert not ierr, f"幂等并发抛错: {ierr}"
   351	        assert sum(1 for _, ins in results if ins) == 1
   352	        assert len({h for h, _ in results}) == 1
   353	        s2 = EventStore(db2)
   354	        try:
   355	            assert s2.head()[1] == 1
   356	        finally:
   357	            s2.close()
   358	    return {"writers": WRITERS, "per_writer": PER_WRITER,
   359	            "total_rows": WRITERS * PER_WRITER,

   366	# STATE-04：事务各点 kill -9 → 全有或全无
   367	# ---------------------------------------------------------------------- #
   368	# 子进程脚本：包装 sqlite3.connect 拦截 execute，在事务的确定性点位对自身发
   369	# SIGKILL（真 kill -9，不是异常模拟）；timer 模式在随机延时后自杀。
   370	_KILL_CHILD_SRC = r'''
   371	import os, signal, sys, threading
   372	import sqlite3
   373	import wenqu_core.store as st
   374	
   375	db, mode, pre, delay = sys.argv[1], sys.argv[2], int(sys.argv[3]), float(sys.argv[4])
   376	
   377	class WrapConn:
   378	    def __init__(self, real):
   379	        self._real = real
   380	        self._begins = 0
   381	    def execute(self, sql, *a):
   382	        cur = self._real.execute(sql, *a)
   383	        up = sql.upper()
   384	        if up.startswith("BEGIN"):
   385	            self._begins += 1
   386	            if self._begins > pre and mode == "after_begin":
   387	                os.kill(os.getpid(), signal.SIGKILL)
   388	        elif mode == "after_insert" and up.startswith("INSERT") and self._begins > pre:
   389	            os.kill(os.getpid(), signal.SIGKILL)
   390	        elif mode == "after_commit" and up.startswith("COMMIT") and self._begins > pre:
   391	            os.kill(os.getpid(), signal.SIGKILL)
   392	        return cur
   393	    def __getattr__(self, name):
   394	        return getattr(self._real, name)
   395	
   396	_orig = sqlite3.connect
   397	sqlite3.connect = lambda *a, **kw: WrapConn(_orig(*a, **kw))
   398	
   399	if mode == "timer":
   400	    threading.Timer(delay, lambda: os.kill(os.getpid(), signal.SIGKILL)).start()
   401	
   402	s = st.EventStore(db)
   403	for i in range(pre):
   404	    s.append({"pre": i})
   405	if mode == "timer":
   406	    for i in range(50):
   407	        s.append({"batch": i})
   408	else:
   409	    s.append({"doomed": True})
   410	print("survived")
   411	'''
   412	
   413	
   414	def _run_kill_child(db, mode, pre, delay=0.0):
   415	    env = dict(os.environ)
   416	    env["PYTHONPATH"] = SYSTEM_DIR + os.pathsep + env.get("PYTHONPATH", "")
   417	    return subprocess.run(
   418	        [sys.executable, "-c", _KILL_CHILD_SRC, db, mode, str(pre), str(delay)],
   419	        capture_output=True, text=True, timeout=60, env=env)

   423	    """注入：事务各点 kill -9（SIGKILL，真崩溃，非异常模拟）。
   424

   426	      after_begin  —— BEGIN IMMEDIATE 刚执行完
   427	      after_insert —— INSERT 已执行、未 COMMIT
   428	      after_commit —— COMMIT 已执行、未返回调用方
   429	    随机点位：timer 模式在 50 条批量写入中随机自杀，重复 8 轮。
   430	    期望（全有或全无）：重开库后哈希链始终可验证；行数只会是
   431	    pre（事务未提交，WAL 回收丢弃）或 pre+1（已提交），绝无撕裂半行；
   432	    批量场景行数落在 [0,50] 且链完好。

   444	                total = store.head()[1]
   445	                verify = store.verify_chain()
   446	                doomed = rows_of(
   447	                    store._conn,

   453	                assert total == pre + 1 and doomed == 1, \
   454	                    f"{mode}: 已提交事务丢失 total={total} doomed={doomed}"
   455	            else:
   456	                assert total == pre and doomed == 0, \
   457	                    f"{mode}: 未提交事务泄漏 total={total} doomed={doomed}"
   458	            assert verify["ok"] is True, f"{mode}: 崩溃后链断裂"
   459	            points[mode] = {"rows": total, "chain_ok": True,
   460	                            "committed": mode == "after_commit"}

   469	            assert p.returncode == -9, f"timer r{round_no}: rc={p.returncode}"
   470	            store = EventStore(db)
   471	            try:
   472	                total = store.head()[1]
   473	                verify = store.verify_chain()
   474	            finally:
   475	                store.close()
   476	            assert 0 <= total <= 50, f"timer r{round_no}: 行数越界 {total}"
   477	            assert verify["ok"] is True, f"timer r{round_no}: 链断裂 @ {total} 行"
   478	            timer_observed.append({"round": round_no, "delay_s": round(delay, 3),
   479	                                   "rows": total, "chain_ok": True})
   480	    return {"deterministic_points": points,
   481	            "timer_rounds": timer_observed,
   482	            "verdict": "每次崩溃后：链完好，行数=已提交事务数（全有或全无）"}
