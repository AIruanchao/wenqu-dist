   261	def extract_project_ids(obj: Dict[str, Any]) -> Dict[str, str]:
   262	    """按 ``PROJECT_FIELDS`` 顺序提取行内的 project 身份字段。
   263	
   264	    只取标量（str/int/float）非空值；dict/list 或空串视为该字段缺席

   267	    """
   268	    out: Dict[str, str] = {}
   269	    for field in PROJECT_FIELDS:
   270	        val = obj.get(field)
   271	        if val is None or isinstance(val, (dict, list, bool)):
   272	            continue
   273	        text = str(val).strip()
   274	        if text:
   275	            out[field] = text
   276	    return out

   343	                for line_no, line in enumerate(fh, start=1):

   356	    # ------------------------------------------------------------------ #
   357	    # 分类：已知别名归一化 / 未知或残缺 -> quarantine
   358	    # ------------------------------------------------------------------ #
   359	    @staticmethod
   360	    def classify(line: str) -> Dict[str, Any]:
   361	        """判定单行账本。
   362	
   363	        返回 ``{"decision": "import"|"quarantine", "status": ..., "reason": ...,
   364	        "line_sha": ..., "project_id": ...}``；import 时 status 为归一化后的
   365	        已知状态、project_id 为行内一致提取的 project 身份（无则 None）；
   366	        MIG-05：多个 project 身份字段值不一致 -> quarantine，reason 形如
   367	        ``project_key_conflict:<fA>=<vA>!=<fB>=<vB>``（值截断 40 字符），
   368	        绝不静默选边导入任一冲突方。
   369	        """
   370	        sha = _line_sha(line)
   371	        base = {"line_sha": sha, "status": None, "reason": None, "project_id": None}
   372	
   373	        try:
   374	            obj = json.loads(line)
   375	        except ValueError:
   376	            return {**base, "decision": "quarantine", "reason": "invalid_json"}
   377	        if not isinstance(obj, dict):
   378	            return {**base, "decision": "quarantine", "reason": "not_a_json_object"}
   379	        for field in REQUIRED_FIELDS:
   380	            if obj.get(field) in (None, ""):
   381	                return {**base, "decision": "quarantine", "reason": f"missing_field:{field}"}
   382	
   383	        raw_status = str(obj["status"]).strip().upper()
   384	        normalized = STATUS_ALIASES.get(raw_status)
   385	        if normalized is None:
   386	            # M2：未知状态绝不猜映射
   387	            return {**base, "decision": "quarantine", "reason": f"unknown_status:{raw_status}"}
   388	
   389	        # M5（MIG-05）：project 身份字段一致性——冲突即显式隔离
   390	        projects = extract_project_ids(obj)
   391	        values = sorted(set(projects.values()))
   392	        if len(values) > 1:
   393	            # 按字段优先级取前两个不一致字段作冲突证据（值截断防长文本）
   394	            it = iter(projects.items())
   395	            fA, vA = next(it)
   396	            fB, vB = next(
   397	                (f, v) for f, v in it if v != projects[fA])
   398	            reason = (f"{PROJECT_KEY_CONFLICT_PREFIX}"
   399	                      f"{fA}={vA[:40]}!={fB}={vB[:40]}")

   401	        project_id = values[0] if values else None

   403	                "project_id": project_id}

   405	    # ------------------------------------------------------------------ #
   406	    # 迁移：经 EventStore.append 写入统一事件表 + 数量守恒校验
   407	    # ------------------------------------------------------------------ #
   408	    def migrate(self) -> Dict[str, Any]:
   409	        """执行迁移并做数量守恒校验（M4）。
   410	
   411	        每行去向恰为三者之一：imported / duplicate / quarantined；
   412	        导入行经公共 ``EventStore.append`` 写入 ``pipeline_events``（单一
   413	        事件表+单一写入 API）；``inserted=False`` 即内容寻址 duplicate
   414	        （等价于旧版 line_sha 唯一冲突）。汇总不守恒 -> 抛
   415	        ConservationError（migration_runs 表同时记账 conserved=0）。
   416	        """
   417	        if not self._inventory:
   418	            self.scan()
   419	
   420	        tally = {
   421	            "files_scanned": len({e["file"] for e in self._inventory}),
   422	            "lines_total": len(self._inventory),
   423	            "imported": 0,
   424	            "duplicates": 0,
   425	            "quarantined": 0,
   426	        }

   428	        status_counts: Dict[str, int] = {}
   429	        per_file: Dict[str, Dict[str, Any]] = {}

   433	        os.makedirs(os.path.dirname(self.sidecar_path) or ".", exist_ok=True)
   434	
   435	        # 单一写入 API：EventStore 负责建表（pipeline_events）、BEGIN
   436	        # IMMEDIATE 事务与哈希链维护；本类不再持有任何第二事件表 DDL。
   437	        store = EventStore(self.sqlite_path)
   438	        try:
   439	            # M1：sidecar 只追加（"a"），源 JSONL 全程只读；
   440	            # sidecar 打不开则整体失败（此时零行导入——先开审计再入账）
   441	            with open(self.sidecar_path, "a", encoding="utf-8") as sidecar:
   442	
   443	                def audit(entry: Dict[str, Any], decision: str, **extra: Any) -> None:
   444	                    record = {
   445	                        "decision": decision,
   446	                        "file": entry["file"],
   447	                        "line_no": entry["line_no"],
   448	                        "line_sha": entry["line_sha"],
   449	                    }

   451	                    sidecar.write(json.dumps(record, ensure_ascii=False, sort_keys=True) + "\n")

   453	                for entry in self._inventory:

   456	                        "lines": 0, "imported": 0, "duplicates": 0, "quarantined": 0})
   457	                    stats["lines"] += 1
   458

   460	                        tally["quarantined"] += 1
   461	                        stats["quarantined"] += 1
   462	                        key = verdict["reason"]

   464	                        audit(entry, "quarantined", reason=key, preview=entry["raw"][:200])
   465	                        continue
   466	
   467	                    # 单一写入 API：与管线控制事件/外部事件同表同链；
   468	                    # inserted=False = 内容寻址 duplicate（本次或既往已入账）；
   469	                    # project_id（若有）随行内容贯通进负载（MIG-05 最小实现）
   470	                    event = migration_event(verdict["line_sha"], verdict["status"],
   471	                                            entry["raw"],
   472	                                            project_id=verdict.get("project_id"))
   473	                    _event_hash, inserted = store.append(event)
   474	                    if not inserted:
   475	                        tally["duplicates"] += 1
   476	                        stats["duplicates"] += 1
   477	                        audit(entry, "duplicate")
   478	                        continue
   479	
   480	                    tally["imported"] += 1
   481	                    stats["imported"] += 1
   482	                    status = verdict["status"]
   483	                    status_counts[status] = status_counts.get(status, 0) + 1
   484	                    audit(entry, "imported", status=status)
   485	
   486	            # 产物链自检：迁移行与库中既有事件（管线控制事件/外部事件）
   487	            # 必须是同一条可验证哈希链——这是「唯一正源」的运行时证明
   488	            store.verify_chain()
   489	
   490	            conserved = (
   491	                tally["lines_total"]
   492	                == tally["imported"] + tally["duplicates"] + tally["quarantined"]
   493	            )
   494	            # run 级记账（bookkeeping 表；非事件正源，不参与哈希链）
   495	            conn = sqlite3.connect(self.sqlite_path)
   496	            try:
   497	                conn.execute("PRAGMA busy_timeout=10000;")
   498	                conn.executescript(_RUNS_SCHEMA)
   499	                conn.execute(
   500	                    "INSERT INTO migration_runs (files_scanned, lines_total, imported, duplicates,"
   501	                    " quarantined, conserved) VALUES (?, ?, ?, ?, ?, ?)",
   502	                    (tally["files_scanned"], tally["lines_total"], tally["imported"],
   503	                     tally["duplicates"], tally["quarantined"], int(conserved)),
   504	                )
   505	                conn.commit()
   506	            finally:
   507	                conn.close()
   508	        finally:
   509	            store.close()
   510	
   511	        if not conserved:
   512	            raise ConservationError(
   513	                f"conservation violated: lines_total={tally['lines_total']} != "
   514	                f"imported({tally['imported']}) + duplicates({tally['duplicates']}) + "
   515	                f"quarantined({tally['quarantined']})"
   516	            )
   517	
   518	        self._last_report = self._build_report(tally, quarantine_reasons, status_counts,
   519	                                               per_file, conserved)
   520	        return self._last_report

   568	def _parse_ts_to_epoch(value: Any) -> Optional[float]:
   569	    """把行内 ts / migrated_at 等时间值解析为 UTC epoch 秒；不可解析返回 None。
   570	
   571	    支持 ISO 8601（含 Z 后缀/无时区按 UTC）、SQLite ``datetime('now')``
   572	    形状、数值 epoch（秒；≥1e12 视为毫秒归一）。不可解析/空值返回
   573	    None——该时间通道缺席，绝不猜时间（M2 精神：缺证据不猜填）。
   574	    """
   575	    if value is None or isinstance(value, bool):
   576	        return None
   577	    if isinstance(value, (int, float)):
   578	        return float(value) / 1000.0 if value >= 1e12 else float(value)
   579	    text = str(value).strip()
   580	    if not text:
   581	        return None
   582	    try:
   583	        num = float(text)
   584	    except ValueError:
   585	        num = None
   586	    if num is not None:
   587	        return num / 1000.0 if num >= 1e12 else num
   588	    iso = (text[:-1] + "+00:00") if text.endswith(("Z", "z")) else text
   589	    parsed: Optional[datetime] = None
   590	    try:
   591	        parsed = datetime.fromisoformat(iso)
   592	    except ValueError:
   593	        for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M:%S.%f",
   594	                    "%Y-%m-%dT%H:%M:%S", "%Y-%m-%dT%H:%M:%S.%f"):
   595	            try:
   596	                parsed = datetime.strptime(text, fmt)
   597	                break
   598	            except ValueError:
   599	                continue
   600	    if parsed is None:
   601	        return None
   602	    if parsed.tzinfo is None:
   603	        parsed = parsed.replace(tzinfo=timezone.utc)
   604	    return parsed.timestamp()

   607	def _row_effective_epoch(payload_text: Optional[str],
   608	                         migrated_at: Any = None) -> Optional[float]:
   609	    """旧行「落账时刻」的保守估计：行内 ts（payload JSON 的 ``ts`` 字段）与
   610	    表元数据 ``migrated_at``（rowid 侧元数据通道）两路取可解析的最大值。
   611	
   612	    取最大是 fail-closed 方向：任一通道声称该行晚于 cutover，即按迟到
   613	    写处理。全部不可解析返回 None（时间证据缺席，由 id 通道兜底）。
   614	    """
   615	    candidates: List[float] = []
   616	    meta = _parse_ts_to_epoch(migrated_at)
   617	    if meta is not None:
   618	        candidates.append(meta)
   619	    if payload_text:
   620	        try:
   621	            obj = json.loads(payload_text)
   622	        except ValueError:
   623	            obj = None
   624	        if isinstance(obj, dict):
   625	            inner = _parse_ts_to_epoch(obj.get("ts"))
   626	            if inner is not None:
   627	                candidates.append(inner)
   628	    return max(candidates) if candidates else None
