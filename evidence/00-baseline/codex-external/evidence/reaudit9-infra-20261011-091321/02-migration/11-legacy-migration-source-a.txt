   261	def extract_project_ids(obj: Dict[str, Any]) -> Dict[str, str]:
   262	    """按 ``PROJECT_FIELDS`` 顺序提取行内的 project 身份字段。
   263	
   264	    只取标量（str/int/float）非空值；dict/list 或空串视为该字段缺席

   267	    """
   268	    out: Dict[str, str] = {}
   269	    for field in PROJECT_FIELDS:
   270	        val = obj.get(field)
   271	        if val is None or isinstance(val, (dict, list, bool)):
   272	            continue
   273	        text = str(val).strip()
   274	        if text:
   275	            out[field] = text
   276	    return out
