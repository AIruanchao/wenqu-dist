   108	def file_sha256(path: str) -> str:
   109	    h = hashlib.sha256()
   110	    with open(path, "rb") as fh:
   111	        for chunk in iter(lambda: fh.read(1 << 20), b""):
   112	            h.update(chunk)
   113	    return h.hexdigest()

   200	def scan_source(source_dir: str) -> Tuple[List[Dict[str, Any]], List[Dict[str, Any]], List[Dict[str, Any]]]:
   201	    """*.jsonl 逐行清单 + 文件元数据 + 跳过文件登记。"""
   202	    if not os.path.isdir(source_dir):
   203	        raise FileNotFoundError(f"source_dir not found: {source_dir}")
   204	    rows: List[Dict[str, Any]] = []
   205	    files_meta: List[Dict[str, Any]] = []
   206	    skipped: List[Dict[str, Any]] = []
   207	    for name in sorted(os.listdir(source_dir)):
   208	        path = os.path.join(source_dir, name)
   209	        if not os.path.isfile(path):
   210	            skipped.append({"file": path, "reason": "not a regular file"})
   211	            continue
   212	        if not name.endswith(".jsonl"):
   213	            skipped.append({"file": path, "reason": "suffix not .jsonl (frozen scan rule)"})
   214	            continue
   215	        with open(path, "rb") as fh:
   216	            data = fh.read()
   217	        text = data.decode("utf-8", errors="replace")
   218	        nonblank = 0
   219	        for line_no, line in enumerate(text.splitlines(), start=1):
   220	            if not line.strip():
   221	                continue
   222	            nonblank += 1
   223	            raw = line.rstrip("\r\n")
   224	            row = {
   225	                "file": path, "file_name": name, "line_no": line_no,
   226	                "raw": raw, "row_id": legacy_event_id(path, line_no, raw),
   227	                "obj": None, "parse_error": None,
   228	            }
   229	            try:
   230	                parsed = json.loads(raw)
   231	                if not isinstance(parsed, dict):
   232	                    row["parse_error"] = "not_a_json_object"
   233	                else:
   234	                    row["obj"] = parsed
   235	            except ValueError as e:
   236	                row["parse_error"] = f"invalid_json:{type(e).__name__}"
   237	            rows.append(row)
   238	        files_meta.append({
   239	            "file": path, "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(),
   240	            "nonblank_lines": nonblank,
   241	            "blank_lines": len(text.splitlines()) - nonblank,
   242	        })
   243	    return rows, files_meta, skipped
