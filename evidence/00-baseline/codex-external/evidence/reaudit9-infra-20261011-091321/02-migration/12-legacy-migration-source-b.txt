   568	def _parse_ts_to_epoch(value: Any) -> Optional[float]:
   569	    """把行内 ts / migrated_at 等时间值解析为 UTC epoch 秒；不可解析返回 None。
   570	
   571	    支持 ISO 8601（含 Z 后缀/无时区按 UTC）、SQLite ``datetime('now')``
   572	    形状、数值 epoch（秒；≥1e12 视为毫秒归一）。不可解析/空值返回
   573	    None——该时间通道缺席，绝不猜时间（M2 精神：缺证据不猜填）。
   574	    """
   575	    if value is None or isinstance(value, bool):
   576	        return None
   577	    if isinstance(value, (int, float)):
   578	        return float(value) / 1000.0 if value >= 1e12 else float(value)
   579	    text = str(value).strip()
   580	    if not text:
   581	        return None
   582	    try:
   583	        num = float(text)
   584	    except ValueError:
   585	        num = None
   586	    if num is not None:
   587	        return num / 1000.0 if num >= 1e12 else num
   588	    iso = (text[:-1] + "+00:00") if text.endswith(("Z", "z")) else text
   589	    parsed: Optional[datetime] = None
   590	    try:
   591	        parsed = datetime.fromisoformat(iso)
   592	    except ValueError:
   593	        for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M:%S.%f",
   594	                    "%Y-%m-%dT%H:%M:%S", "%Y-%m-%dT%H:%M:%S.%f"):
   595	            try:
   596	                parsed = datetime.strptime(text, fmt)
   597	                break
   598	            except ValueError:
   599	                continue

   601	        return None
   602	    if parsed.tzinfo is None:
   603	        parsed = parsed.replace(tzinfo=timezone.utc)
