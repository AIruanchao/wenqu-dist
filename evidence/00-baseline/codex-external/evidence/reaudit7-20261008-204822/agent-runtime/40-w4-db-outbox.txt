#!/bin/zsh
set -u
SRC="$HOME/.wenqu/state/scheduler.db"
DST="/private/tmp/reaudit7-scheduler-copy-$$.db"
cp "$SRC" "$DST"
chmod u+w "$DST"
echo "source_sha=$(shasum -a 256 "$SRC" | awk '{print $1}')"
echo "copy_sha=$(shasum -a 256 "$DST" | awk '{print $1}')"
echo '=== tables/schema ==='
sqlite3 "$DST" '.tables'
sqlite3 "$DST" '.schema scheduler_outbox'
echo '=== outbox selected rows (body excluded) ==='
sqlite3 -header -column "$DST" "SELECT message_id,channel,severity,subject,status,attempts,max_attempts,next_attempt_at,sent_at,last_error,created_at,meta_json FROM scheduler_outbox ORDER BY created_at,message_id;"
echo '=== outbox aggregate ==='
sqlite3 -header -column "$DST" "SELECT channel,status,count(*) AS n,min(created_at) AS first_created,max(sent_at) AS last_sent,max(attempts) AS max_attempts FROM scheduler_outbox GROUP BY channel,status ORDER BY channel,status;"
echo '=== explicit dead letters ==='
sqlite3 -header -column "$DST" "SELECT message_id,channel,severity,subject,status,attempts,max_attempts,last_error FROM scheduler_outbox WHERE status='DEAD_LETTER' ORDER BY created_at;"
echo '=== heartbeat and runs latest ==='
sqlite3 "$DST" '.schema scheduler_runs'
sqlite3 -header -column "$DST" "SELECT run_id,job_id,scheduled_at,started_at,finished_at,execution_status,policy_verdict,actual_exit_code FROM scheduler_runs ORDER BY started_at DESC LIMIT 12;"
rm -f "$DST"
