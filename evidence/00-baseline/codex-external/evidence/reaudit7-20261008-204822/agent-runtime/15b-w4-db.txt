#!/bin/zsh
set -o pipefail
SRC="$HOME/.wenqu/state/scheduler.db"; DST=/tmp/reaudit7-scheduler-db-copy2.sqlite
cp -p "$SRC" "$DST"
stat -f 'SRC_SIZE=%z SRC_MTIME=%Sm' -t '%Y-%m-%dT%H:%M:%S%z' "$SRC"

echo SKIPS
sqlite3 -header -column "$DST" 'select job_id,count(*) as n,min(created_at) as first,max(created_at) as last from scheduler_skips group by job_id;'
echo LAST_LOGS
