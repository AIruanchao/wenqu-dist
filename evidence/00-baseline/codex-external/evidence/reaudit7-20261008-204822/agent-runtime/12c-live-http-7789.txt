#!/bin/zsh

for spec in 'GET /' 'GET /api/v1/ping' 'GET /api/v1/components' 'GET /api/v1/health' 'GET /api/health' 'POST /api/decide' 'POST /api/v1/decide' 'PUT /api/v1/health' 'OPTIONS /api/v1/ping'; do

 body=$(mktemp /tmp/reaudit7-http.XXXXXX); hdr=$(mktemp /tmp/reaudit7-hdr.XXXXXX)
 if [[ $method == POST ]]; then
   code=$(curl -sS --max-time 8 -X "$method" -H 'Content-Type: application/json' --data '{}' -D "$hdr" -o "$body" -w '%{http_code}' "http://127.0.0.1:7789$reqpath"); rc=$?
