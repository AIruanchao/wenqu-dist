     4	W10 观察窗样本（~/.wenqu/observation/samples/YYYY-MM-DD.json）：

    17	注：modular server（system/dashboard/server.py）的 health 契约路径是
    18	  /api/v1/health（未版本化 /api/health 已废止，返回 404 API_VERSION_REQUIRED）。

    66	def find_bin(name):
    67	    found = shutil.which(name)
    68	    if found:
    69	        return found
    70	    for d in _BIN_FALLBACK_DIRS:
    71	        cand = os.path.join(d, name)
    72	        if os.path.isfile(cand) and os.access(cand, os.X_OK):
    73	            return cand

    77	def run_cmd(argv, timeout, cwd=None):
    78	    """跑一条命令，永不抛异常。返回 {"returncode", "stdout", "stderr", "error"}。"""
    79	    try:
    80	        p = subprocess.run(
    81	            argv,
    82	            capture_output=True,
    83	            text=True,
    84	            timeout=timeout,
    85	            cwd=cwd,
    86	        )
    87	        return {
    88	            "returncode": p.returncode,
    89	            "stdout": (p.stdout or "").strip(),
    90	            "stderr": (p.stderr or "").strip()[:500],
    91	            "error": None,
    92	        }
    93	    except Exception as e:  # FileNotFoundError / TimeoutExpired / OSError
    94	        return {
    95	            "returncode": None,
    96	            "stdout": "",
    97	            "stderr": "",
    98	            "error": str(e)[:300],
    99	        }

   102	def http_get_json(url, timeout=HTTP_TIMEOUT):
   103	    """GET 一个 URL 并解析 JSON，永不抛异常。返回 {"status", "json", "error"}。"""
   104	    req = urlrequest.Request(url, headers={"Accept": "application/json", "User-Agent": UA})
   105	    try:
   106	        with _OPENER.open(req, timeout=timeout) as resp:
   107	            body = resp.read(262144).decode("utf-8", "replace")
   108	            status = getattr(resp, "status", None) or resp.getcode()
   109	        parsed = None
   110	        if body:
   111	            try:
   112	                parsed = json.loads(body)
   113	            except ValueError:
   114	                parsed = None
   115	        return {"status": status, "json": parsed, "error": None if parsed is not None else "body_not_json"}
   116	    except urlerror.HTTPError as e:
   117	        body = ""
   118	        try:
   119	            body = e.read(262144).decode("utf-8", "replace")
   120	        except Exception:
   121	            pass
   122	        parsed = None
   123	        try:
   124	            parsed = json.loads(body) if body else None
   125	        except ValueError:
   126	            parsed = None
   127	        return {"status": e.code, "json": parsed, "error": "http_" + str(e.code)}
   128	    except Exception as e:
   129	        return {"status": None, "json": None, "error": str(e)[:200]}

   132	def http_status_only(url, timeout=2.0):
   133	    """轻探：返回 HTTP 状态码 int；连接层失败返回 None（用于就绪轮询）。"""
   134	    req = urlrequest.Request(url, headers={"User-Agent": UA})
   135	    try:
   136	        with _OPENER.open(req, timeout=timeout) as resp:
   137	            return getattr(resp, "status", None) or resp.getcode()
   138	    except urlerror.HTTPError as e:
   139	        return e.code
   140	    except Exception:
   141	        return None

   144	def safe(fn):
   145	    """单探针兜底：任何异常转为 {"error": ...}，不中断采样。"""
   146	    try:
   147	        val = fn()
   148	        return val if val is not None else {"error": "probe_returned_null"}
   149	    except Exception as e:
   150	        return {"error": str(e)[:300]}

   155	def probe_host():
   156	    uname_bin = find_bin("uname")
   157	    if not uname_bin:
   158	        return {"error": "uname not found"}
   159	    r = run_cmd([uname_bin, "-a"], timeout=10)
   160	    return r["stdout"] if r["returncode"] == 0 and r["stdout"] else {"error": r["error"] or ("rc=" + str(r["returncode"]) + " " + r["stderr"][:120])}

   163	def probe_repo():
   164	    git_bin = find_bin("git")
   165	    if not git_bin:
   166	        return {"error": "git not found"}
   167	    head_r = run_cmd([git_bin, "rev-parse", "HEAD"], timeout=15, cwd=REPO)
   168	    dirty_r = run_cmd([git_bin, "status", "--porcelain"], timeout=15, cwd=REPO)
   169	    head = head_r["stdout"] if head_r["returncode"] == 0 else None
   170	    dirty_files = [ln for ln in dirty_r["stdout"].splitlines() if ln.strip()] if dirty_r["returncode"] == 0 else None
   171	    return {
   172	        "path": REPO,
   173	        "head": head,
   174	        "head_error": None if head else (head_r["error"] or head_r["stderr"][:200]),
   175	        "dirty": (len(dirty_files) > 0) if dirty_files is not None else None,
   176	        "dirty_files": dirty_files if dirty_files is not None else None,
   177	        "dirty_error": None if dirty_files is not None else dirty_r["error"],
   178	    }

   181	def probe_ci_latest_main():
   182	    gh_bin = find_bin("gh")
   183	    if not gh_bin:
   184	        return None
   185	    r = run_cmd(
   186	        [gh_bin, "run", "list", "--branch", "main", "--limit", "1",
   187	         "--json", "status,conclusion,databaseId,createdAt,headSha,displayTitle,workflowName"],
   188	        timeout=GH_TIMEOUT,
   189	        cwd=REPO,
   190	    )
   191	    if r["returncode"] != 0 or not r["stdout"]:
   192	        return {"error": r["error"] or ("rc=" + str(r["returncode"]) + " " + r["stderr"][:200])}
   193	    try:
   194	        runs = json.loads(r["stdout"])
   195	    except ValueError:
   196	        return {"error": "gh output not json"}
   197	    if not isinstance(runs, list) or not runs:
   198	        return {"error": "no runs returned"}
   199	    return runs[0]

   203	    launchctl_bin = find_bin("launchctl")

   241	def stop_process_tree(proc):
   242	    """必杀：先 SIGTERM 进程组，3s 不退再 SIGKILL。返回终止方式。"""
   243	    if proc.poll() is not None:
   244	        return "already_exited_rc=" + str(proc.returncode)
   245	    try:
   246	        os.killpg(os.getpgid(proc.pid), signal.SIGTERM)
   247	    except OSError:
   248	        try:
   249	            proc.terminate()
   250	        except OSError:
   251	            pass
   252	    try:
   253	        proc.wait(timeout=3)
   254	        return "term"
   255	    except subprocess.TimeoutExpired:
   256	        try:
   257	            os.killpg(os.getpgid(proc.pid), signal.SIGKILL)
   258	        except OSError:
   259	            try:
   260	                proc.kill()
   261	            except OSError:
   262	                pass
   263	        try:
   264	            proc.wait(timeout=3)
   265	        except subprocess.TimeoutExpired:
   266	            pass
   267	        return "kill"

   270	def compare_health(legacy, modular):
   271	    """key 集合相等性 / score 相等性 / 键差。不可比时字段记 null，不猜。"""
   272	    out = {
   273	        "key_sets_equal": None,
   274	        "score_equal": None,
   275	        "only_legacy": None,
   276	        "only_modular": None,
   277	        "legacy_score": None,
   278	        "modular_score": None,
   279	        "note": None,
   280	    }
   281	    lj = legacy.get("json") if isinstance(legacy, dict) and legacy.get("status") == 200 else None
   282	    mj = modular.get("json") if isinstance(modular, dict) and modular.get("status") == 200 else None
   283	    if not isinstance(lj, dict):
   284	        out["note"] = "legacy /api/health 无 200 JSON，无法比对"
   285	        return out
   286	    if not isinstance(mj, dict):
   287	        out["legacy_score"] = lj.get("score") if isinstance(lj.get("score"), (int, float)) else None
   288	        out["note"] = "modular health 无 200 JSON（当前为结构化错误态），无法比对"
   289	        return out
   290	    lk = set(lj.keys())
   291	    mk = set(mj.keys())
   292	    out["only_legacy"] = sorted(lk - mk)
   293	    out["only_modular"] = sorted(mk - lk)
   294	    out["key_sets_equal"] = lk == mk
   295	    ls = lj.get("score")
   296	    ms = mj.get("score")
   297	    out["legacy_score"] = ls if isinstance(ls, (int, float)) and not isinstance(ls, bool) else None
   298	    out["modular_score"] = ms if isinstance(ms, (int, float)) and not isinstance(ms, bool) else None
   299	    if out["legacy_score"] is not None and out["modular_score"] is not None:
   300	        out["score_equal"] = out["legacy_score"] == out["modular_score"]
   301	    else:
   302	        out["note"] = "至少一方缺 numeric score 字段，score 不比对"
   303	    return out

   306	def probe_shadow():
   307	    if not os.path.isfile(SERVER_PY):
   308	        return {"error": "server.py not found: " + SERVER_PY}
   309	    port = pick_free_port()
   310	    base_url = "http://" + HOST_LOCAL + ":" + str(port)
   311	    ping_url = base_url + "/api/v1/ping"
   312	    v1_health_url = base_url + "/api/v1/health"
   313	    unversioned_health_url = base_url + "/api/health"
   314	    server_log_path = os.path.join(LOGS_DIR, "shadow-server-" + date.today().isoformat() + ".log")
   315	
   316	    result = {
   317	        "port": port,
   318	        "server_pid": None,
   319	        "ready": False,
   320	        "ready_seconds": None,
   321	        "server_log": server_log_path,
   322	        "terminated": None,
   323	    }
   324	    logf = open(server_log_path, "ab")
   325	    try:
   326	        proc = subprocess.Popen(
   327	            [sys.executable, SERVER_PY, "--port", str(port)],
   328	            start_new_session=True,
   329	            stdout=logf,
   330	            stderr=subprocess.STDOUT,
   331	            cwd=REPO,
   332	        )
   333	        result["server_pid"] = proc.pid
   334	        t0 = time.monotonic()
   335	        while (time.monotonic() - t0) < SHADOW_READY_TIMEOUT:
   336	            if proc.poll() is not None:
   337	                result["ready_error"] = "server exited early rc=" + str(proc.returncode)
   338	                break
   339	            code = http_status_only(ping_url, timeout=2.0)
   340	            if code is not None:
   341	                result["ready"] = True
   342	                result["ready_http"] = code
   343	                result["ready_seconds"] = round(time.monotonic() - t0, 2)
   344	                break
   345	            time.sleep(0.5)
   346	        else:
   347	            result["ready_error"] = "not ready within " + str(int(SHADOW_READY_TIMEOUT)) + "s"
   348	
   349	        legacy = probe_legacy_health()
   350	        v1_health = http_get_json(v1_health_url)
   351	        unversioned_health = http_get_json(unversioned_health_url)
   352	
   353	        # 比对主源：优先 200 JSON 的 v1；否则退回未版本化路径的 200 JSON；
   354	        # 都没有时保留 v1 原始结果（含 503/404 结构化错误），如实观测。
   355	        modular = v1_health
   356	        modular_path = "/api/v1/health"
   357	        if not (v1_health.get("status") == 200 and isinstance(v1_health.get("json"), dict)):
   358	            if unversioned_health.get("status") == 200 and isinstance(unversioned_health.get("json"), dict):
   359	                modular = unversioned_health
   360	                modular_path = "/api/health"
   361	
   362	        result["modular_health"] = modular
   363	        result["modular_health_path"] = modular_path
   364	        result["modular_health_v1_raw"] = {"status": v1_health.get("status"), "error": v1_health.get("error")}
   365	        result["modular_health_unversioned_raw"] = {"status": unversioned_health.get("status"), "error": unversioned_health.get("error")}
   366	        result["legacy_health"] = legacy
   367	        result["compare"] = compare_health(legacy, modular)
   368	    finally:
   369	        if result.get("server_pid") is not None:
   370	            result["terminated"] = stop_process_tree(proc)
   371	        logf.close()
   372	    return result

   377	def write_json_atomic(path, payload):
   378	    tmp = path + ".tmp"
   379	    with open(tmp, "w", encoding="utf-8") as f:
   380	        json.dump(payload, f, ensure_ascii=False, indent=2, sort_keys=False)
   381	        f.write("\n")

   385	def rotate(dirpath):
   386	    """按文件名日期删除超过 KEEP_DAYS 天的样本。"""
   387	    cutoff = date.today() - timedelta(days=KEEP_DAYS)
   388	    removed = []
   389	    try:
   390	        names = os.listdir(dirpath)
   391	    except OSError:
   392	        return removed
   393	    for name in names:
   394	        if not name.endswith(".json"):
   395	            continue
   396	        try:
   397	            d = datetime.strptime(name[:-len(".json")], "%Y-%m-%d").date()
   398	        except ValueError:
   399	            continue
   400	        if d < cutoff:
   401	            try:
   402	                os.remove(os.path.join(dirpath, name))
   403	                removed.append(name)
   404	            except OSError:
   405	                pass
   406	    return removed
