   265	def build_catalog():
   266	    """构建 131 个 ID 的冻结目录并做完整性自检。"""
   267	    catalog = {}
   268	    for (prefix, num), (injection, expectation) in ID_DEFS.items():
   269	        test_id = f"{prefix}-{num:02d}"
   270	        req, ac, wps, ev_class = PREFIX_META[prefix]
   271	        catalog[test_id] = {
   272	            "prefix": prefix,
   273	            "section": SECTION_OF.get((prefix, num), "20.5"),
   274	            "injection": injection,
   275	            "expectation": expectation,
   276	            "req": req,
   277	            "ac": ac,
   278	            "work_packages": wps,
   279	            "evidence_class": ev_class,
   280	        }
   281	    problems = []
   282	    if len(catalog) != EXPECTED_TOTAL:
   283	        problems.append(f"目录数量 {len(catalog)} != {EXPECTED_TOTAL}")
   284	    # 前缀分母核对（§25.1 范围展开）
   285	    expected_counts = {
   286	        "GATE": 12, "AUTH": 6, "ACT": 6, "MRG": 7, "STATE": 11, "MIG": 5, "DB": 6,
   287	        "SC": 5, "ADV": 2, "TEN": 8, "PERF": 6, "SCH": 7, "ALT": 2, "BAK": 1,
   288	        "REL": 3, "STOP": 3, "RUN": 3, "CI": 5, "COND": 6, "RISK": 1, "STA": 2,
   289	        "DEP": 9, "RB": 1, "UI": 7, "HLT": 1, "DOC": 4, "CAP": 1, "DR": 1,
   290	    }
   291	    actual_counts = {}
   292	    for tid in catalog:
   293	        actual_counts[tid.split("-")[0]] = actual_counts.get(tid.split("-")[0], 0) + 1
   294	    if actual_counts != expected_counts:
   295	        for k in sorted(set(expected_counts) | set(actual_counts)):
   296	            if expected_counts.get(k) != actual_counts.get(k):
   297	                problems.append(
   298	                    f"前缀 {k}: 期望 {expected_counts.get(k)} 实际 {actual_counts.get(k)}")
   299	    # 编号连续性（CI 特例：05~09）
   300	    ranges = {"CI": (5, 9)}
   301	    for tid in catalog:
   302	        prefix, num = tid.split("-")[0], int(tid.split("-")[1])
   303	        lo, hi = ranges.get(prefix, (1, expected_counts[prefix]))
   304	        if not (lo <= num <= hi):
   305	            problems.append(f"ID {tid} 超出定义范围 {lo}~{hi}")
   306	    dup = EXPECTED_TOTAL - len(catalog)
   307	    if dup:
   308	        problems.append(f"疑似重复/缺失 {dup} 项")
   309	    return catalog, problems

   312	def cross_check_plan(plan_path, catalog):
   313	    """可选：解析方案 §20 表格，与冻结目录做 ID 集合等式校验 + 逐 ID 语义比对。
   314	
   315	    F4-TRC-001 根修（Codex 第四轮实锤）：旧版只比 ID 集合——保留 ID 不动、
   316	    只改该行「注入/期望」文本的语义篡改全绿。现从 §20.1～§20.5 表格逐行解析
   317	    (ID, 注入, 期望)，与冻结目录（ID_DEFS 转录）做归一化语义比对：
   318	    去首尾空白、markdown 反引号剥离、全半角标点归一、连续空白折叠。
   319	    任一 ID 语义漂移 → semantic problems → --check exit 2。
   320	    """
   321	    try:
   322	        with open(plan_path, encoding="utf-8") as f:
   323	            text = f.read()
   324	    except OSError as e:
   325	        return {"ok": False, "error": f"读取方案失败: {e}", "plan_ids": None}
   326	    body = text
   327	    # 只取 §20～§21 之间的正文，避免 §25 表格里的范围写法（GATE-01～12）污染集合
   328	    m20 = re.search(r"^# 20\.", body, re.M)
   329	    m21 = re.search(r"^# 21\.", body, re.M)
   330	    if m20 and m21:
   331	        body = body[m20.start():m21.start()]
   332	    ids = set(re.findall(r"\b([A-Z]{2,5}-\d{2})\b", body))
   333	    plan_ids = {i for i in ids if not re.match(r"^(FND|W\d)", i)}
   334	    frozen = set(catalog)
   335	    # ---- 逐 ID 语义比对（F4-TRC-001）------------------------------------
   336	    semantic_problems = []
   337	    row_re = re.compile(
   338	        r"^\|\s*([A-Z]{2,5}-\d{2})\s*\|\s*([^|]+?)\s*\|\s*([^|]+?)\s*\|\s*$")
   339	    plan_semantics = {}
   340	    for line in body.splitlines():
   341	        mm = row_re.match(line)
   342	        if mm:
   343	            plan_semantics[mm.group(1)] = (mm.group(2), mm.group(3))
   344	    if not plan_semantics:
   345	        semantic_problems.append("§20 表格解析出 0 行数据行（格式漂移？）")
   346	    for tid in sorted(frozen):
   347	        if tid not in plan_semantics:
   348	            # 集合等式已另行报告；这里防「ID 在正文他处出现但表格行被删」的缝隙
   349	            semantic_problems.append(f"§20 表格缺 {tid} 行（或行格式无法解析）")
   350	            continue
   351	        plan_inj, plan_exp = plan_semantics[tid]
   352	        frozen_inj = catalog[tid]["injection"]
   353	        frozen_exp = catalog[tid]["expectation"]
   354	        if _norm_semantic(plan_inj) != _norm_semantic(frozen_inj):
   355	            semantic_problems.append(
   356	                f"{tid} 注入语义漂移: 方案={plan_inj!r} 冻结={frozen_inj!r}")
   357	        if _norm_semantic(plan_exp) != _norm_semantic(frozen_exp):
   358	            semantic_problems.append(
   359	                f"{tid} 期望语义漂移: 方案={plan_exp!r} 冻结={frozen_exp!r}")
   360	    for tid in sorted(set(plan_semantics) - frozen):
   361	        semantic_problems.append(f"§20 表格行 {tid} 不在冻结目录中")
   362	    return {
   363	        "ok": plan_ids == frozen and not semantic_problems,
   364	        "plan_only": sorted(plan_ids - frozen),
   365	        "frozen_only": sorted(frozen - plan_ids),
   366	        "plan_ids_count": len(plan_ids),
   367	        "frozen_count": len(frozen),
   368	        "semantic_ok": not semantic_problems,
   369	        "semantic_problems": semantic_problems,
   370	        "semantic_rows_parsed": len(plan_semantics),
   371	    }

   374	# 全角→半角标点归一表（语义比对用；只归一标点，不动字母/数字/汉字）
   375	_FW_PUNCT = {
   376	    "，": ",", "、": ",", "。": ".", "？": "?", "！": "!", "：": ":",
   377	    "；": ";", "（": "(", "）": ")", "～": "~", "「": '"', "」": '"',
   378	    "『": '"', "』": '"', "“": '"', "”": '"', "‘": "'", "’": "'",
   379	    "【": "[", "】": "]", "《": "<", "》": ">", "　": " ",
   380	}
   381	
   382	
   383	def _norm_semantic(s):
   384	    """§20 语义归一化：去 markdown 反引号、全半角标点归一、折叠空白、去首尾。"""
   385	    if not isinstance(s, str):
   386	        return "" if s is None else str(s)
   387	    out = s.replace("`", "")
   388	    out = "".join(_FW_PUNCT.get(ch, ch) for ch in out)
   389	    out = re.sub(r"\s+", " ", out)
   390	    return out.strip()

   393	def _norm_meta(s):
   394	    """§25.1 表格与冻结元数据的归一化：全角～→半角~、去首尾空白。"""
   395	    return s.replace("～", "~").strip()

   398	def cross_check_plan_251(plan_path, catalog):
   399	    """真解析方案 §25.1 映射表：范围展开成 ID 集合 + 逐前缀四列元数据对账。
   400	
   401	    防的是第三轮验收实锤的假绿：篡改 §25.1 映射（换 REQ/AC/工作包/证据类）
   402	    时旧版工具根本不解析该表却声称"§25.1 集合等式"。
   403	    """
   404	    try:
   405	        with open(plan_path, encoding="utf-8") as f:
   406	            text = f.read()
   407	    except OSError as e:
   408	        return {"ok": False, "error": f"读取方案失败: {e}"}
   409	    m = re.search(r"^## 25\.1\b", text, re.M)
   410	    if not m:
   411	        return {"ok": False, "error": "方案中未找到 §25.1 小节"}
   412	    rest = text[m.start():]
   413	    m_end = re.search(r"^# 2[6-9]\.|^---\s*$", rest[m.end() - m.start():], re.M)
   414	    section = rest if not m_end else rest[: m.end() - m.start() + m_end.start()]
   415	    row_re = re.compile(
   416	        r"^\|\s*([A-Z]{2,5})-(\d{2})(?:～(\d{2}))?\s*\|([^|]+)\|([^|]+)\|([^|]+)\|")
   417	    expanded, meta, problems = set(), {}, []
   418	    for line in section.splitlines():
   419	        mm = row_re.match(line)
   420	        if not mm:
   421	            continue
   422	        prefix, lo = mm.group(1), int(mm.group(2))
   423	        hi = int(mm.group(3)) if mm.group(3) else lo
   424	        req_ac, work_packages, evidence_class = mm.group(4), mm.group(5), mm.group(6)
   425	        parts = [p.strip() for p in req_ac.split("/", 1)]
   426	        if len(parts) != 2 or not parts[0].startswith("REQ-") or not parts[1].startswith("AC-"):
   427	            problems.append(f"§25.1 行 {prefix}-{lo:02d}～{hi:02d} 需求/AC 列格式非法: {req_ac.strip()!r}")
   428	            continue
   429	        for n in range(lo, hi + 1):
   430	            tid = f"{prefix}-{n:02d}"
   431	            if tid in expanded:
   432	                problems.append(f"§25.1 重复展开 ID: {tid}")
   433	            expanded.add(tid)
   434	        meta[prefix] = (_norm_meta(req_ac.replace(" / ", "/", 1)),
   435	                        _norm_meta(work_packages), _norm_meta(evidence_class))
   436	    if not expanded:
   437	        return {"ok": False, "error": "§25.1 表格解析出 0 行（格式漂移？）"}
   438	    frozen = set(catalog)
   439	    if expanded != frozen:
   440	        problems.append(f"§25.1 展开集合≠冻结目录: 仅方案={sorted(expanded - frozen)} 仅冻结={sorted(frozen - expanded)}")
   441	    for prefix, plan_meta in sorted(meta.items()):
   442	        frozen_meta = PREFIX_META.get(prefix)
   443	        if frozen_meta is None:
   444	            problems.append(f"§25.1 前缀 {prefix} 不在冻结元数据中")
   445	            continue
   446	        norm_frozen = (_norm_meta(frozen_meta[0] + "/" + frozen_meta[1]),
   447	                       _norm_meta(frozen_meta[2]), _norm_meta(frozen_meta[3]))
   448	        if plan_meta != norm_frozen:
   449	            problems.append(f"§25.1 前缀 {prefix} 四列元数据与冻结值不符: 方案={plan_meta} 冻结={norm_frozen}")
   450	    return {"ok": not problems, "problems": problems,
   451	            "expanded_count": len(expanded), "frozen_count": len(frozen)}

   457	def iter_test_files():
   458	    for root_rel in SCAN_ROOTS:

   469	TESTISH_NAME = re.compile(r"(^test_|_self_test$|^_?selftest|self_test)", re.I)
   470	
   471	
   472	def scan_python(path, wanted_forms):
   473	    """在 .py 中定位 ID 出现点，归属到 test 型函数才算 implemented。"""
   474	    try:
   475	        with open(path, encoding="utf-8") as f:
   476	            src = f.read()
   477	        tree = ast.parse(src)
   478	    except (OSError, SyntaxError, UnicodeDecodeError) as e:
   479	        return {}, f"parse 失败: {e}"
   480	    funcs = []  # (name, lineno, end_lineno)
   481	    for node in ast.walk(tree):
   482	        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
   483	            funcs.append((node.name, node.lineno, node.end_lineno or node.lineno))
   484	    lines = src.splitlines()
   485	    hits = {}
   486	    for form in wanted_forms:
   487	        for i, line in enumerate(lines, start=1):
   488	            if form in line:
   489	                owner = None
   490	                for name, lo, hi in funcs:
   491	                    if lo <= i <= hi:
   492	                        owner = name
   493	                        break
   494	                if owner is not None and TESTISH_NAME.search(owner):
   495	                    hits.setdefault(form, []).append((owner, i))
   496	    return hits, None

   499	def scan_shell(path, wanted_forms):
   500	    try:
   501	        with open(path, encoding="utf-8") as f:
   502	            lines = f.read().splitlines()
   503	    except (OSError, UnicodeDecodeError) as e:
   504	        return {}, f"read 失败: {e}"
   505	    hits = {}
   506	    for form in wanted_forms:
   507	        for i, line in enumerate(lines, start=1):
   508	            if form in line and not line.strip().startswith("#"):
   509	                hits.setdefault(form, []).append(("(script-case)", i))
   510	    return hits, None

   513	def collect_test_matches(catalog):
   514	    """返回 {test_id: {"test_file":..., "test_function":..., "line":...}}（专属命中）。"""
   515	    # 归一化形态：GATE-01 / GATE_01（函数名下划线写法）
   516	    id_forms = {}
   517	    for tid in catalog:
   518	        prefix, num = tid.split("-")
   519	        id_forms[tid] = [f"{prefix}-{num}", f"{prefix}_{num}", f"test_{prefix}_{num}"]
   520	    matches = {}
   521	    scanned = []
   522	    all_forms = sorted({f for forms in id_forms.values() for f in forms})
   523	    for path in iter_test_files():
   524	        rel = os.path.relpath(path, REPO_ROOT)
   525	        scanned.append(rel)
   526	        try:
   527	            with open(path, encoding="utf-8", errors="ignore") as f:
   528	                content = f.read()
   529	        except OSError:
   530	            continue
   531	        present = [f for f in all_forms if f in content]
   532	        if not present:
   533	            continue
   534	        if path.endswith(".py"):
   535	            hits, err = scan_python(path, present)
   536	        else:
   537	            hits, err = scan_shell(path, present)
   538	        if err:
   539	            continue
   540	        # form -> test_id 反查
   541	        for tid, forms in id_forms.items():
   542	            for form in forms:
   543	                if form in hits:
   544	                    owner, line = hits[form][0]
   545	                    matches.setdefault(tid, {
   546	                        "test_file": rel, "test_function": owner,
   547	                        "line": line, "match_kind": "dedicated",
   548	                    })
   549	                    break
   550	    return matches, scanned

   553	def collect_evidence_matches(catalog):
   554	    """evidence/ 下含 ID 的非 README 工件（排除本工具输出）。"""
   555	    matches = {}
   556	    if not os.path.isdir(EVIDENCE_ROOT):
   557	        return matches
   558	    for dirpath, dirnames, filenames in os.walk(EVIDENCE_ROOT):
   559	        for fn in filenames:
   560	            if fn == "README.md" or fn in SELF_OUTPUT_NAMES or fn.endswith(".sha256"):
   561	                continue
   562	            p = os.path.join(dirpath, fn)
   563	            rel = os.path.relpath(p, REPO_ROOT)
   564	            try:
   565	                if os.path.getsize(p) > 2_000_000:
   566	                    continue
   567	                with open(p, encoding="utf-8", errors="ignore") as f:
   568	                    content = f.read()
   569	            except OSError:
   570	                continue
   571	            for tid in catalog:
   572	                if tid in content:
   573	                    matches.setdefault(tid, rel)
   574	    return matches

   588	def main():
   589	    ap = argparse.ArgumentParser(description="§20 131 验收 ID 追踪矩阵生成器")
   590	    ap.add_argument("--json", default=os.path.join(EVIDENCE_ROOT, "traceability-matrix.json"),
   591	                    help="输出 JSON 路径（默认 evidence/traceability-matrix.json）")

   594	                                         "evidence", "00-baseline", "codex-external", "方案.md"),
   595	                    help="方案文件路径（默认仓内快照 evidence/00-baseline/codex-external/方案.md；存在则做 ID 集合交叉校验）")
   596	    ap.add_argument("--check", action="store_true",
   597	                    help="目录完整性/集合校验失败时 exit 2")
   598	    args = ap.parse_args()
   599

   601	    integrity_ok = not problems

   603	    plan_check = None
   604	    if args.plan and os.path.isfile(args.plan):
   605	        plan_check = cross_check_plan(args.plan, catalog)
   606	        if not plan_check.get("ok", False):
   607	            if (plan_check.get("plan_only") or plan_check.get("frozen_only")
   608	                    or plan_check.get("error")):
   609	                problems.append(f"方案交叉校验不一致: {plan_check.get('plan_only')}/{plan_check.get('frozen_only')}")
   610	            for p in (plan_check.get("semantic_problems") or [])[:20]:
   611	                # F4-TRC-001：逐 ID 语义漂移必须计入 problems（--check exit 2）
   612	                problems.append(f"§20 语义漂移: {p}")
   613	        plan_check_251 = cross_check_plan_251(args.plan, catalog)
   614	        if not plan_check_251.get("ok", False):
   615	            problems.append("§25.1 映射交叉校验不一致: " + "; ".join(
   616	                plan_check_251.get("problems") or [plan_check_251.get("error", "未知")]))
   617	
   618	    direct, scanned = collect_test_matches(catalog)
   619	    evidence_map = collect_evidence_matches(catalog)
   620	
   621	    matrix = {}
   622	    n_impl = n_proxy = 0
   623	    for tid in sorted(catalog):
   624	        meta = catalog[tid]
   625	        entry = {
   626	            "implemented": False,
   627	            "test_file": "",
   628	            "evidence_path": "",
   629	            "match_type": "missing",
   630	            "test_function": "",
   631	            "section": meta["section"],
   632	            "injection": meta["injection"],
   633	            "expectation": meta["expectation"],
   634	            "req": meta["req"],
   635	            "ac": meta["ac"],
   636	            "work_packages": meta["work_packages"],
   637	            "evidence_class": meta["evidence_class"],
   638	        }
   639	        if tid in direct:
   640	            entry.update(implemented=True, match_type="dedicated",
   641	                         test_file=direct[tid]["test_file"],
   642	                         test_function=direct[tid]["test_function"],
   643	                         line=direct[tid]["line"])
   644	            n_impl += 1
   645	        elif tid in PROXY_ALIASES:
   646	            pa = PROXY_ALIASES[tid]
   647	            entry.update(implemented=False, match_type="proxy_partial",
   648	                         test_file=pa["test_file"],
   649	                         test_function=pa["test_function"],
   650	                         proxy_note=pa["note"])
   651	            n_proxy += 1
   652	        if tid in evidence_map:
   653	            entry["evidence_path"] = evidence_map[tid]
   654	        matrix[tid] = entry
   655	
   656	    total = len(matrix)
   657	    missing = total - n_impl - n_proxy
