===== system/wenqu_core/gate_aggregator.py =====
     1	# -*- coding: utf-8 -*-
     2	"""统一 Gate 聚合器（Codex W3B；P0-3 二次加固：双契约消费 + dashboard 输出对齐；
     3	F4-GATE-001 根修：v2 输入结构必须严格——add() 前置全量镜像校验）。
     4	
     5	消费流水线各 station 产出的 station-result JSON，聚合为整线统一的 gate 判定。
     6	
     7	## 输入契约一（station-result-v2，正式契约——schemas/station-result-v2.schema.json）
     8	
     9	判别为 v2（`schema_version == "2.0"`，或出现 v2 正式字段组合 station_id/
    10	execution_status/policy_verdict）时，**结构必须严格**：add() 先经
    11	:func:`mirror_validate_station_result_v2`（schema 的无外部依赖镜像校验器）
    12	全量校验 required/类型/enum/pattern/const/allOf-if-then 跨字段语义——
    13	任何 schema-invalid 输入一律记 malformed、该站强制 BLOCKED、
    14	technical_eligible=false（Codex 第四轮实锤：残缺 v2 曾被判 PASS）。
    15	字段语义保持双契约：v2 正式字段 + legacy 兼容字段并存消费。
    16	
    17	    {
    18	      "schema_version":     "2.0",
    19	      "station_id":         2,             # 必填：站点编号（0-7）；站点键 = str(station_id)
    20	      "execution_status":   "COMPLETED",   # 必填：COMPLETED/ERROR/BLOCKED/TIMEOUT/CANCELLED
    21	      "policy_verdict":     "PASS",        # 必填：PASS/FAIL/CONDITIONAL/NOT_APPLICABLE/NOT_EVALUATED
    22	      "identity": {                        # 必填：证据身份绑定（C2/C5 的数据源）
    23	        "commit_sha":       "<40-hex>",    #   与构造器 target_sha 比对
    24	        "environment":      "staging",     #   与构造器 environment 比对
    25	        ...
    26	      },
    27	    }
    28	
    29	v2 verdict 归一（fail-closed）：
    30	    PASS                        -> PASS
    31	    CONDITIONAL / NOT_APPLICABLE-> CONDITIONAL（软——不得上绿，需人工处置）
    32	    FAIL / NOT_EVALUATED        -> BLOCKED（硬失败）
    33	    execution_status != COMPLETED（ERROR/BLOCKED/TIMEOUT/CANCELLED）
    34	                                -> 整站硬失败（铁律 2：执行未完成不得记 PASS）
    35	
    36	## 输入契约二（legacy 旧字段，向后兼容——按 entry 的 input_version 区分）::
    37	
    38	    {
    39	      "station":     "lint",        # 站点名（required_stations 成员或旁路站点）
    40	      "outcome":     "PASS",        # 见别名表；未知/缺失一律硬失败
    41	      "sha":         "<full-sha>",  # 亦接受 target_sha / commit_sha
    42	      "environment": "staging",     # 须与构造器 environment 一致
    43	    }
    44	
    45	outcome 别名表（大小写不敏感）：
    46	    PASS/PASSED/OK/SUCCESS/GREEN                            -> PASS
    47	    CONDITIONAL/WARN/WARNING/SOFT_PASS/PASS_WITH_WARNINGS   -> CONDITIONAL
    48	    FAIL/FAILED/FAILURE/ERROR/BLOCKED/TIMEOUT/未知/缺失     -> 硬失败（整线 BLOCKED）
    49	
    50	## 硬约束（不可降级）
    51	    C1 任一 required 站非 PASS -> 整线非 PASS；
    52	    C2 任一站点证据 SHA 与 target_sha 不一致（或缺失、无法绑定目标）-> BLOCKED；
    53	    C3 分母缩水：required 站未全部上报 -> BLOCKED；空 required_stations 同样 BLOCKED；
    54	    C4 CONDITIONAL != PASS：条件通过永不映射为绿判定 / GitHub success；
    55	    C5 环境不一致 / 结构残缺 / 同站冲突证据 -> BLOCKED（宁可阻断，不猜证据）。
    56	
    57	## 输出契约（P0-3 修：对齐 dashboard/server.py read_gate_aggregate 强制校验）
    58	
    59	aggregate() 输出除原有字段外必须包含：
    60	    generated_at   ISO8601 UTC（含 Z 后缀，dashboard 以 fromisoformat 解析并查 TTL）
    61	    policy_verdict 整线判定（PASS/CONDITIONAL/BLOCKED；dashboard 要求
    62	                   policy_verdict/overall/verdict 三者至少其一）
    63	端到端保证：合法 v2 输入聚合出的 JSON 喂给 dashboard 的 read_gate_aggregate()
    64	必须 ACCEPT（不触发 AGGREGATOR_MALFORMED/AGGREGATOR_STALE）。
    65	
    66	technical_eligible 语义：True 仅表示技术硬门（C2/C3/C5 及“无任何站点硬失败”）
    67	全部通过；与 aggregate_outcome == PASS 不等价——CONDITIONAL 时 technical_eligible
    68	可为 True，但放行判断必须且只能以 aggregate_outcome == PASS 为准。
    69	
    70	GitHub 状态映射（to_github_status）：
    71	    PASS        -> state=success  / conclusion=success
    72	    CONDITIONAL -> state=failure  / conclusion=action_required  # 条件项须人工处置，绝不上绿
    73	    BLOCKED     -> state=failure  / conclusion=failure
    74	"""

   119	# ====================================================================== #
   120	# station-result-v2 镜像校验器（F4-GATE-001 根修，2026-10-08）
   121	# ====================================================================== #
   122	# schemas/station-result-v2.schema.json 的无外部依赖结构镜像——语义必须与
   123	# schema 文件逐条同步（required/类型/enum/pattern/const/additionalProperties/
   124	# allOf-if-then 跨字段）。system/tests/test_gate_schema_hardening.py 内置
   125	# 一致性自测：对正负样例组，本镜像与 jsonschema+FormatChecker（严格
   126	# date-time 检查注册后）结论必须一致。改 schema 文件时必须同步改这里。
   127	#
   128	# 与 jsonschema 的两处刻意差异（均有一致性自测护栏）：
   129	#   1. date-time：镜像无条件做严格 ISO-8601 解析（jsonschema 的 FormatChecker
   130	#      在未安装 rfc3339-validator 时对 date-time 空转——自测按仓内既有惯例
   131	#      注册严格检查后对齐）；
   132	#   2. PASS 的 scanned==denominator：schema 侧以 const 级联（1..64）机器判定
   133	#      （draft-07 无法表达跨字段数值比较），分母>64 时不误拒合法全扫；镜像
   134	#      侧做全量无界等式判定（缩水无论多大必拒）。
   135	_MIRROR_TOP_KEYS = frozenset({
   136	    "schema_version", "run_id", "station_id", "attempt_id",
   137	    "execution_status", "policy_verdict", "identity", "tool", "execution",
   138	    "coverage", "finding_ids", "artifacts",
   139	})
   140	_MIRROR_TOP_REQUIRED = (
   141	    "schema_version", "run_id", "station_id", "attempt_id",
   142	    "execution_status", "policy_verdict", "identity", "tool", "execution",
   143	)
   144	_MIRROR_IDENTITY_REQUIRED = ("commit_sha", "environment", "scope_hash")
   145	_MIRROR_IDENTITY_OPTIONAL = (
   146	    "project_id", "ruleset_hash", "data_config_hash", "lockfile_hash",
   147	)
   148	_MIRROR_EXEC_REQUIRED = (
   149	    "argv_digest", "started_at", "ended_at", "actual_exit_code",
   150	    "expected_exit_set", "assertion_verdict",
   151	)
   152	_MIRROR_RUN_ID_RE = re.compile(r"^[a-zA-Z0-9_-]+$")
   153	_MIRROR_SHA40_RE = re.compile(r"^[0-9a-f]{40}$")
   154	_MIRROR_CAS_RE = re.compile(r"^sha256:[0-9a-f]{64}$")
   155	_MIRROR_EXEC_STATUS = frozenset(
   156	    {"COMPLETED", "ERROR", "BLOCKED", "TIMEOUT", "CANCELLED"})
   157	_MIRROR_VERDICTS = frozenset(
   158	    {"PASS", "FAIL", "CONDITIONAL", "NOT_APPLICABLE", "NOT_EVALUATED"})
   159	_MIRROR_ASSERTION = frozenset({"PASS", "FAIL", "ERROR"})
   160	_MIRROR_EXEC_HARD = frozenset({"ERROR", "BLOCKED", "TIMEOUT", "CANCELLED"})
   161	# F4-SCHEMA-001 #5（station 侧）：执行非完成态只允许阻断类裁决
   162	_MIRROR_HARD_EXEC_VERDICTS = frozenset({"FAIL", "NOT_EVALUATED", "NOT_APPLICABLE"})
   163	
   164	
   165	def _mirror_int(value: Any) -> bool:
   166	    """schema "integer"：bool 不是 integer（JSON Schema 语义）。"""
   167	    return isinstance(value, int) and not isinstance(value, bool)
   168	
   169	
   170	def _mirror_iso_datetime(value: Any) -> bool:
   171	    """format: date-time（严格 ISO-8601；Z/z 后缀按 UTC 接受）。"""
   172	    if not isinstance(value, str):
   173	        return False
   174	    text = value[:-1] + "+00:00" if value.endswith(("Z", "z")) else value
   175	    try:
   176	        datetime.fromisoformat(text)
   177	        return True
   178	    except ValueError:
   179	        return False

   182	def mirror_validate_station_result_v2(result: Any) -> List[str]:
   183	    """station-result-v2.schema.json 的镜像校验；返回违例清单（空 == 合法）。
   184	
   185	    无外部依赖（纯标准库）；GateAggregator.add() 对每个 v2 条目前置调用，
   186	    schema-invalid 一律记 malformed 并强制该站 BLOCKED（fail-closed）。
   187	    """
   188	    errors: List[str] = []
   189	
   190	    def _err(msg: str) -> None:
   191	        errors.append(msg)
   192	
   193	    if not isinstance(result, Mapping):
   194	        return ["station result must be a JSON object"]
   195	
   196	    # -- 顶层 required / additionalProperties:false ------------------------
   197	    for key in _MIRROR_TOP_REQUIRED:
   198	        if key not in result:
   199	            _err(f"required: {key}")
   200	    extra = sorted(set(result) - _MIRROR_TOP_KEYS)
   201	    if extra:
   202	        _err(f"additionalProperties not allowed: {extra}")
   203	
   204	    # -- 顶层标量 -----------------------------------------------------------
   205	    if result.get("schema_version") != "2.0":
   206	        _err("schema_version must be const '2.0'")
   207	    run_id = result.get("run_id")
   208	    if not isinstance(run_id, str) or not _MIRROR_RUN_ID_RE.match(run_id):
   209	        _err("run_id must be a string matching ^[a-zA-Z0-9_-]+$")
   210	    sid = result.get("station_id")
   211	    if not _mirror_int(sid) or not 0 <= sid <= 7:
   212	        _err("station_id must be an integer in [0,7]")
   213	    if not isinstance(result.get("attempt_id"), str):
   214	        _err("attempt_id must be a string")
   215	    if result.get("execution_status") not in _MIRROR_EXEC_STATUS:
   216	        _err("execution_status must be one of "
   217	             f"{sorted(_MIRROR_EXEC_STATUS)}")
   218	    if result.get("policy_verdict") not in _MIRROR_VERDICTS:
   219	        _err("policy_verdict must be one of "
   220	             f"{sorted(_MIRROR_VERDICTS)}")
   221	
   222	    # -- identity -----------------------------------------------------------
   223	    identity = result.get("identity")
   224	    if not isinstance(identity, Mapping):
   225	        _err("identity must be an object")
   226	    else:
   227	        for key in _MIRROR_IDENTITY_REQUIRED:
   228	            if key not in identity:
   229	                _err(f"identity.required: {key}")
   230	        sha = identity.get("commit_sha")
   231	        if not isinstance(sha, str) or not _MIRROR_SHA40_RE.match(sha):
   232	            _err("identity.commit_sha must be a string matching "
   233	                 "^[0-9a-f]{40}$")
   234	        if not isinstance(identity.get("environment"), str):
   235	            _err("identity.environment must be a string")
   236	        if not isinstance(identity.get("scope_hash"), str):
   237	            _err("identity.scope_hash must be a string")
   238	        for key in _MIRROR_IDENTITY_OPTIONAL:
   239	            if key in identity and not isinstance(identity[key], str):
   240	                _err(f"identity.{key} must be a string")
   241	
   242	    # -- tool ----------------------------------------------------------------
   243	    tool = result.get("tool")
   244	    if not isinstance(tool, Mapping):
   245	        _err("tool must be an object")
   246	    else:
   247	        for key in ("name", "version"):
   248	            if key not in tool:
   249	                _err(f"tool.required: {key}")
   250	            elif not isinstance(tool[key], str):
   251	                _err(f"tool.{key} must be a string")
   252	        if "digest" in tool and not isinstance(tool["digest"], str):
   253	            _err("tool.digest must be a string")
   254	
   255	    # -- execution -----------------------------------------------------------
   256	    execution = result.get("execution")
   257	    if not isinstance(execution, Mapping):
   258	        _err("execution must be an object")
   259	    else:
   260	        for key in _MIRROR_EXEC_REQUIRED:
   261	            if key not in execution:
   262	                _err(f"execution.required: {key}")
   263	        if not isinstance(execution.get("argv_digest"), str):
   264	            _err("execution.argv_digest must be a string")
   265	        for key in ("started_at", "ended_at"):
   266	            if key in execution and not _mirror_iso_datetime(execution[key]):
   267	                _err(f"execution.{key} must be an ISO-8601 date-time")
   268	        if "timeout_s" in execution and (
   269	                not _mirror_int(execution["timeout_s"])
   270	                or execution["timeout_s"] < 1):
   271	            _err("execution.timeout_s must be an integer >= 1")
   272	        if not _mirror_int(execution.get("actual_exit_code")):
   273	            _err("execution.actual_exit_code must be an integer")
   274	        expected = execution.get("expected_exit_set")
   275	        if not isinstance(expected, list) or not all(
   276	                _mirror_int(x) for x in expected):
   277	            _err("execution.expected_exit_set must be an array of integers")
   278	        if execution.get("assertion_verdict") not in _MIRROR_ASSERTION:
   279	            _err("execution.assertion_verdict must be one of "
   280	                 f"{sorted(_MIRROR_ASSERTION)}")
   281	        if "cwd_digest" in execution and not isinstance(
   282	                execution["cwd_digest"], str):
   283	            _err("execution.cwd_digest must be a string")
   284	
   285	    # -- coverage（可选；PASS 时必填并升级约束）------------------------------
   286	    coverage = result.get("coverage") if "coverage" in result else None
   287	    if "coverage" in result:
   288	        if not isinstance(coverage, Mapping):
   289	            _err("coverage must be an object")
   290	        else:
   291	            for key in ("denominator", "scanned"):
   292	                if key not in coverage:
   293	                    _err(f"coverage.required: {key}")
   294	                elif not _mirror_int(coverage[key]) or coverage[key] < 0:
   295	                    _err(f"coverage.{key} must be an integer >= 0")
   296	            if "exclusions" in coverage and (
   297	                    not isinstance(coverage["exclusions"], list)
   298	                    or not all(isinstance(x, str)
   299	                               for x in coverage["exclusions"])):
   300	                _err("coverage.exclusions must be an array of strings")
   301	
   302	    # -- finding_ids / artifacts ---------------------------------------------
   303	    if "finding_ids" in result and (
   304	            not isinstance(result["finding_ids"], list)
   305	            or not all(isinstance(x, str) for x in result["finding_ids"])):
   306	        _err("finding_ids must be an array of strings")
   307	    artifacts = result.get("artifacts") if "artifacts" in result else None
   308	    if "artifacts" in result:
   309	        if not isinstance(artifacts, list):
   310	            _err("artifacts must be an array")
   311	        else:
   312	            for idx, item in enumerate(artifacts):
   313	                if not isinstance(item, Mapping):
   314	                    _err(f"artifacts[{idx}] must be an object")
   315	                    continue
   316	                for key in ("cas_digest", "size"):
   317	                    if key not in item:
   318	                        _err(f"artifacts[{idx}].required: {key}")
   319	                digest = item.get("cas_digest")
   320	                if not isinstance(digest, str) or not _MIRROR_CAS_RE.match(digest):
   321	                    _err(f"artifacts[{idx}].cas_digest must match "
   322	                         "^sha256:[0-9a-f]{64}$")
   323	                if not _mirror_int(item.get("size")) or item["size"] < 1:
   324	                    _err(f"artifacts[{idx}].size must be an integer >= 1")
   325	
   326	    # -- allOf 跨字段 #1：PASS 分支（schema PASS if-then 的全量镜像）----------
   327	    if result.get("policy_verdict") == "PASS":
   328	        if "coverage" not in result:
   329	            _err("PASS requires coverage")
   330	        if "artifacts" not in result:
   331	            _err("PASS requires artifacts")
   332	        if isinstance(coverage, Mapping) and _mirror_int(
   333	                coverage.get("denominator")) and _mirror_int(
   334	                coverage.get("scanned")):
   335	            if coverage["denominator"] < 1:
   336	                _err("PASS coverage.denominator must be >= 1")
   337	            if coverage["scanned"] < 1:
   338	                _err("PASS coverage.scanned must be >= 1")
   339	            # F4-SCHEMA-001 #2（分母缩水）：PASS 必须 scanned == denominator
   340	            # （schema 侧 const 级联 1..64 的无界镜像）
   341	            if coverage["scanned"] != coverage["denominator"]:
   342	                _err("PASS coverage.scanned("
   343	                     f"{coverage['scanned']}) != denominator("
   344	                     f"{coverage['denominator']}) — denominator shrinkage")
   345	        if isinstance(artifacts, list) and len(artifacts) < 1:
   346	            _err("PASS requires non-empty artifacts (minItems 1)")
   347	        if isinstance(execution, Mapping):
   348	            if "actual_exit_code" in execution and (
   349	                    execution["actual_exit_code"] != 0):
   350	                _err("PASS requires execution.actual_exit_code == 0")
   351	            if "assertion_verdict" in execution and (
   352	                    execution["assertion_verdict"] != "PASS"):
   353	                _err("PASS requires execution.assertion_verdict == PASS")
   354	        # F4-SCHEMA-001 #5（station 侧）：PASS 必须执行完成
   355	        if result.get("execution_status") != "COMPLETED":
   356	            _err("PASS requires execution_status COMPLETED")
   357	
   358	    # -- allOf 跨字段 #2：执行非完成态只允许阻断类裁决 ------------------------
   359	    # （F4-SCHEMA-001 #5：TIMEOUT+CONDITIONAL / TIMEOUT+PASS 等双轴洗白全拒）
   360	    if result.get("execution_status") in _MIRROR_EXEC_HARD and (
   361	            result.get("policy_verdict") is not None
   362	            and result.get("policy_verdict")
   363	            not in _MIRROR_HARD_EXEC_VERDICTS):
   364	        _err(f"execution_status {result['execution_status']} forbids "
   365	             f"policy_verdict {result.get('policy_verdict')} "
   366	             "(only FAIL/NOT_EVALUATED allowed)")
   367	
   368	    return errors

   371	class GateAggregator:
   372	    """消费 station-result JSON（v2 正式契约优先、legacy 兼容），输出整线统一 gate 判定。
   373	
   374	    用法::
   375	
   376	        agg = GateAggregator(required_stations={"2", "7"},   # v2 站点键 = str(station_id)
   377	                             target_sha="abc...", environment="staging")
   378	        agg.add(station_result_v2).add(...)
   379	        verdict = agg.aggregate()          # 含 generated_at/policy_verdict（dashboard 契约）
   380	        status = agg.to_github_status()    # {"state": ..., "conclusion": ..., "description": ...}
   381	
   382	    required_stations 兼容 int 站点号（自动归一为字符串键，如 2 -> "2"）。
   383	    """
   384	
   385	    def __init__(self, required_stations: Set[str], target_sha: str, environment: str) -> None:
   386	        target_sha = str(target_sha or "").strip().lower()
   387	        if not target_sha:
   388	            raise ValueError("target_sha is required: gate evidence must be bound to a SHA")
   389	        if not isinstance(environment, str) or not environment:
   390	            raise ValueError("environment is required: gate evidence must be bound to an environment")
   391	
   392	        self.required_stations: frozenset = frozenset(
   393	            str(s).strip() for s in required_stations
   394	            if isinstance(s, (str, int)) and not isinstance(s, bool) and str(s).strip()
   395	        )
   396	        self.target_sha: str = target_sha
   397	        self.environment: str = environment

   410	    @staticmethod
   411	    def _normalize_outcome(raw: Any) -> str:
   412	        """归一化 legacy outcome；未知/缺失一律按 BLOCKED（硬失败），绝不猜成好结果。"""
   413	        token = str(raw if raw is not None else "").strip().upper()
   414	        if token in _PASS_ALIASES:
   415	            return PASS
   416	        if token in _CONDITIONAL_ALIASES:
   417	            return CONDITIONAL
   418	        return BLOCKED

   420	    @staticmethod
   421	    def _is_v2(station_result: Mapping) -> bool:
   422	        """v2 判别：显式 schema_version=2.0，或出现 v2 正式字段组合。"""
   423	        if str(station_result.get("schema_version", "")).strip() == _V2_SCHEMA_VERSION:
   424	            return True
   425	        return "station_id" in station_result and (
   426	            "policy_verdict" in station_result or "execution_status" in station_result
   427	        )

   429	    def _parse_v2(self, station_result: Mapping) -> Dict[str, Any]:
   430	        """解析 v2 条目为内部登记结构；结构必须严格（F4-GATE-001 根修）。
   431	
   432	        先过 mirror_validate_station_result_v2（schema 全量镜像）：schema-invalid
   433	        记一条 consolidated malformed（含违例明细），该站 outcome 强制 BLOCKED；
   434	        identity/sha/environment 仍按现有方式尽力提取，供 C2/C5 证据绑定裁决
   435	        （缺什么补什么违规——绝不因结构残缺而漏记账）。
   436	        """
   437	        schema_errors = mirror_validate_station_result_v2(station_result)
   438	        sid = station_result.get("station_id")
   439	        if not isinstance(sid, int) or isinstance(sid, bool):
   440	            self._malformed.append({
   441	                "reason": "missing_field:station_id",
   442	                "entry_keys": sorted(map(str, station_result.keys())),
   443	            })
   444	            return {}
   445	        station = str(sid)
   446	
   447	        if schema_errors:
   448	            self._malformed.append({
   449	                "reason": "schema_invalid: " + schema_errors[0],
   450	                "station": station,
   451	                "schema_violations": schema_errors[:8],
   452	                "schema_violation_count": len(schema_errors),
   453	            })
   454	
   455	        execution_status = str(station_result.get("execution_status") or "").strip().upper()
   456	        policy_verdict = str(station_result.get("policy_verdict") or "").strip().upper()
   457	
   458	        identity = station_result.get("identity")
   459	        sha = ""
   460	        environment = None
   461	        if isinstance(identity, Mapping):
   462	            raw_sha = identity.get("commit_sha")
   463	            if isinstance(raw_sha, str) and raw_sha.strip():
   464	                sha = raw_sha.strip().lower()
   465	            env_raw = identity.get("environment")
   466	            if isinstance(env_raw, str):
   467	                environment = env_raw
   468	        else:
   469	            # identity 缺失/非对象——schema_invalid 已记账（镜像 required:identity），
   470	            # 这里继续提取其余字段并保持 C5 记账（宽松读不炸，聚合时 BLOCKED）
   471	            if not schema_errors:
   472	                self._malformed.append({
   473	                    "reason": "missing_field:identity" if identity is None
   474	                    else "malformed_field:identity",
   475	                    "station": station,
   476	                })
   477	
   478	        # 铁律 2：执行未完成（ERROR/BLOCKED/TIMEOUT/CANCELLED）→ 整站硬失败
   479	        if schema_errors:
   480	            # F4-GATE-001：schema-invalid 输入不具可信形态——该站一律硬失败
   481	            outcome = BLOCKED
   482	            outcome_raw = "SCHEMA_INVALID"
   483	        elif execution_status in _V2_EXEC_HARD_FAIL:
   484	            outcome = BLOCKED
   485	            outcome_raw = f"{execution_status}/{policy_verdict or '-'}"
   486	        elif execution_status == "COMPLETED":
   487	            outcome = _V2_VERDICT_MAP.get(policy_verdict, BLOCKED)
   488	            outcome_raw = f"{execution_status}/{policy_verdict or '-'}"
   489	        else:
   490	            # execution_status 缺失/未知（宽松读不炸）——fail-closed 硬失败
   491	            outcome = BLOCKED
   492	            outcome_raw = f"{execution_status or '-'}/{policy_verdict or '-'}"
   493	
   494	        if not schema_errors:
   495	            # 枚举合法性已由镜像覆盖（schema_invalid 时不再重复记账，避免噪声）
   496	            if execution_status and execution_status not in _V2_EXECUTION_STATUS:
   497	                self._malformed.append({
   498	                    "reason": f"invalid_field:execution_status:{execution_status}",
   499	                    "station": station,
   500	                })
   501	            if policy_verdict and policy_verdict not in _V2_POLICY_VERDICTS:
   502	                self._malformed.append({
   503	                    "reason": f"invalid_field:policy_verdict:{policy_verdict}",
   504	                    "station": station,
   505	                })
   506	
   507	        return {
   508	            "station": station,
   509	            "outcome": outcome,
   510	            "outcome_raw": outcome_raw,
   511	            "sha": sha,
   512	            "environment": environment,
   513	            "required": station in self.required_stations,
   514	            "input_version": "v2",
   515	            "execution_status": execution_status or None,
   516	            "policy_verdict_raw": policy_verdict or None,
   517	            "schema_valid": not schema_errors,
   518	        }

   545	            "environment": environment,
   546	            "required": station in self.required_stations,

   553	    def add(self, station_result: dict) -> "GateAggregator":
   554	        """登记一条 station-result（v2 或 legacy 自动判别）；返回 self 以便链式调用。
   555	
   556	        v2 条目先过镜像校验器（F4-GATE-001）：schema-invalid 一律记 malformed
   557	        并强制该站 BLOCKED——聚合输出必为 BLOCKED、technical_eligible=false。
   558	
   559	        重复站点：内容完全一致视为幂等上报（忽略）；不一致视为冲突证据，
   560	        两者都会在 aggregate() 时以 BLOCKED 或备注形式体现，绝不静默择优。
   561	        """
   562	        if not isinstance(station_result, Mapping):
   563	            self._malformed.append(
   564	                {"reason": "not_a_json_object", "entry": repr(station_result)[:200]}
   565	            )
   566	            return self
   567	
   568	        if self._is_v2(station_result):
   569	            entry = self._parse_v2(station_result)
   570	        else:
   571	            entry = self._parse_legacy(station_result)
   572	        if not entry:
   573	            return self  # 已记 malformed，aggregate() 时 BLOCKED
   574	        station = entry["station"]
   575	
   576	        previous = self._stations.get(station)
   577	        if previous is not None:
   578	            identical = (
   579	                previous["outcome"] == entry["outcome"]
   580	                and previous["sha"] == entry["sha"]
   581	                and previous["environment"] == entry["environment"]
   582	            )
   583	            if identical:
   584	                self._duplicates.append(station)
   585	            else:
   586	                self._conflicts.append({"station": station, "first": previous, "second": entry})
   587	            return self  # 首条为准；冲突已记账，aggregate() 时 BLOCKED
   588	
   589	        self._stations[station] = entry
   590	
   591	        # C2：SHA 证据绑定（缺失同样无法绑定 -> 违规）
   592	        if entry["sha"] != self.target_sha:
   593	            self._sha_violations.append({
   594	                "station": station,
   595	                "reason": "missing_sha" if not entry["sha"] else "sha_mismatch",
   596	                "sha": entry["sha"] or None,
   597	                "expected": self.target_sha,
   598	            })
   599	        # C5：环境证据绑定（缺失同样无法绑定 -> 违规）
   600	        if entry["environment"] != self.environment:
   601	            self._env_violations.append({
   602	                "station": station,
   603	                "reason": "missing_environment" if entry["environment"] is None else "environment_mismatch",
   604	                "environment": entry["environment"],
   605	                "expected": self.environment,
   606	            })
   607	        return self

   612	    def aggregate(self) -> Dict[str, Any]:
   613	        """聚合为统一 gate 判定；不修改登记状态（generated_at 为当下时刻）。"""
   614	        reasons: List[str] = []
   615	        hard = False
   616	
   617	        def block(reason: str) -> None:
   618	            nonlocal hard
   619	            hard = True
   620	            reasons.append(reason)
   621	
   622	        # 1) C2/C5：证据完整性硬门
   623	        if self._sha_violations:
   624	            detail = "; ".join(f"{v['station']}:{v['reason']}" for v in self._sha_violations)
   625	            block(f"sha_evidence_violation [{detail}]")
   626	        if self._env_violations:
   627	            detail = "; ".join(f"{v['reason']}" for v in self._env_violations)
   628	            stations = ",".join(sorted({v["station"] for v in self._env_violations}))
   629	            block(f"environment_evidence_violation [{stations}] ({detail})")
   630	        if self._malformed:
   631	            block(f"malformed_station_result x{len(self._malformed)} "
   632	                  f"[first={self._malformed[0]['reason']}]")
   633	        if self._conflicts:
   634	            names = ",".join(sorted({c["station"] for c in self._conflicts}))
   635	            block(f"duplicate_station_conflict [{names}]")
   636	
   637	        # 2) C3：分母缩水（required 站未全部上报）；空 required 同样拒绝（无真空 PASS）
   638	        reported = set(self._stations)
   639	        missing_required = sorted(self.required_stations - reported)
   640	        bypass_reported = sorted(reported - self.required_stations)
   641	        if missing_required or not self.required_stations:
   642	            if not self.required_stations:
   643	                block("denominator_shrinkage: no required stations configured "
   644	                      "(refusing vacuous PASS)")
   645	            else:
   646	                block("denominator_shrinkage: missing required stations [" +
   647	                      ",".join(missing_required) + "]")
   648	
   649	        # 3) 站点硬失败（required 或旁路站一视同仁；未知 outcome 已归入硬失败）
   650	        hard_failed = sorted(s for s, e in self._stations.items() if e["outcome"] == BLOCKED)
   651	        if hard_failed:
   652	            block("station_hard_failure [" + ",".join(hard_failed) + "]")
   653	
   654	        # 4) C4：条件项（软，不 block 但整线不得为 PASS）
   655	        conditional = sorted(s for s, e in self._stations.items() if e["outcome"] == CONDITIONAL)
   656	        if conditional:
   657	            reasons.append("conditional_stations [" + ",".join(conditional) + "]")
   658	
   659	        if hard:
   660	            outcome = BLOCKED
   661	        elif conditional:
   662	            outcome = CONDITIONAL
   663	        else:
   664	            outcome = PASS
   665	
   666	        if not self.required_stations:
   667	            reason = "BLOCKED: no required stations configured — refusing vacuous PASS (P0-3 fix)"
   668	        elif outcome == PASS:
   669	            reason = f"all {len(self.required_stations)} required stations PASS"
   670	        else:
   671	            reason = reasons[0] if reasons else outcome
   672	
   673	        stations_summary = {
   674	            s: {
   675	                "outcome": e["outcome"],
   676	                "required": e["required"],
   677	                "sha_bound": e["sha"] == self.target_sha,
   678	                "environment_ok": e["environment"] == self.environment,
   679	                "input_version": e["input_version"],
   680	                "schema_valid": e.get("schema_valid", True),
   681	            }
   682	            for s, e in sorted(self._stations.items())
   683	        }
   684	
   685	        return {
   686	            # dashboard/server.py read_gate_aggregate() 强制要求的契约字段（P0-3 修）
   687	            "generated_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
   688	            "policy_verdict": outcome,
   689	            # 原有字段（保持稳定）
   690	            "aggregate_outcome": outcome,
   691	            "technical_eligible": (not hard) and outcome == PASS,
   692	            "reason": reason,
   693	            "reasons": reasons,
   694	            "counts": {
   695	                "required_total": len(self.required_stations),
   696	                "required_reported": len(self.required_stations & reported),
   697	                "required_missing": len(missing_required),
   698	                "bypass_stations": len(bypass_reported),
   699	                "duplicates_ignored": len(self._duplicates),
   700	                "conflicts": len(self._conflicts),
   701	                "malformed": len(self._malformed),
   702	            },
   703	            "stations": stations_summary,
   704	            "target_sha": self.target_sha,
   705	            "environment": self.environment,
   706	        }

   220	def _gate_exit_code(outcome: str) -> int:
   221	    """aggregate_outcome -> CLI 退出码（F4-GATE-001 契约）。
   222	
   223	    PASS=0；CONDITIONAL=1（软条件项绝不上绿、需人工处置，映射 GitHub
   224	    failure/action_required）；BLOCKED=2（证据残缺/分母缩水/绑定失败——
   225	    fail-closed）；其余（不应出现）一律 2 不放水。
   226	    """
   227	    return {"PASS": 0, "CONDITIONAL": 1, "BLOCKED": 2}.get(outcome, 2)

   230	def _gate_error_json(reason: str, **extra: Any) -> int:
   231	    """ERROR(3) 路径：gate 工具自身无法执行（输入不可读/无法绑定身份）。
   232	
   233	    仍打印一个 aggregate 形状的 JSON（policy_verdict=ERROR）供脚本消费，
   234	    退出码固定 3——与证据性 BLOCKED(2) 严格区分。
   235	    """
   236	    payload: Dict[str, Any] = {
   237	        "generated_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
   238	        "policy_verdict": "ERROR",
   239	        "aggregate_outcome": "ERROR",
   240	        "technical_eligible": False,
   241	        "reason": reason,
   242	        "reasons": [reason],
   243	    }
   244	    payload.update(extra)
   245	    print(f"wenquctl: gate ERROR: {reason}", file=sys.stderr)
   246	    print(json.dumps(payload, ensure_ascii=False))
   247	    return 3

   250	def cmd_gate(args: argparse.Namespace) -> int:
   251	    """gate 子命令——读入 station-result-v2 JSON，聚合为整线 gate 判定。
   252	
   253	    每份结果逐个过内置结构校验器（validate_station_result——schema 的可执行
   254	    等价物）+ GateAggregator 严格登记（宽松读、聚合 fail-closed）；
   255	    target_sha/environment 缺省从首个可绑定结果的 identity 推导（跨 SHA/环境
   256	    混报由聚合器 C2/C5 判 BLOCKED），--required 缺省=全部已上报站（防分母缩水
   257	    真空绿）。结构校验失败的输入强制整线 BLOCKED（绝不静默跳过——跳过即分母
   258	    缩水）。退出码映射见 _gate_exit_code / _gate_error_json。
   259	    """
   260	    # 1. 展开 --results：文件直取；目录按名序取 *.json。
   261	    paths: List[str] = []
   262	    for target in args.results:
   263	        if os.path.isdir(target):
   264	            names = sorted(n for n in os.listdir(target) if n.endswith(".json"))
   265	            if not names:
   266	                return _gate_error_json(
   267	                    f"results 目录无 .json 结果文件: {target}",
   268	                    inputs={"paths": 0, "docs": 0, "load_errors": 0,
   269	                            "schema_invalid": 0})
   270	            paths.extend(os.path.join(target, n) for n in names)
   271	        elif os.path.isfile(target):
   272	            paths.append(target)
   273	        else:
   274	            return _gate_error_json(
   275	                f"results 路径不存在: {target}",
   276	                inputs={"paths": 0, "docs": 0, "load_errors": 0,
   277	                        "schema_invalid": 0})
   278	
   279	    # 2. 载入 JSON（不可读/非 JSON 记账为输入错误——ERROR 3，操作者给错了文件）。
   280	    docs: List[Any] = []
   281	    load_errors: List[Dict[str, str]] = []
   282	    for path in paths:
   283	        try:
   284	            with open(path, "r", encoding="utf-8") as fh:
   285	                doc = json.load(fh)
   286	        except (OSError, json.JSONDecodeError) as exc:
   287	            load_errors.append({"source": path, "error": str(exc)})
   288	            continue
   289	        if isinstance(doc, list):  # 单文件多结果（数组信封）
   290	            docs.extend(doc)
   291	        else:
   292	            docs.append(doc)
   293	    if load_errors:
   294	        # 输入面不完整（文件不可读/非 JSON）——绝不带着缺口继续聚合出 PASS。
   295	        return _gate_error_json(
   296	            f"results 输入不可读/非 JSON x{len(load_errors)} "
   297	            f"[first={load_errors[0]['source']}: {load_errors[0]['error'][:120]}]",
   298	            inputs={"paths": len(paths), "docs": len(docs),
   299	                    "load_errors": len(load_errors), "schema_invalid": 0},
   300	            load_errors=load_errors)
   301	
   302	    # 3. 推导 target_sha / environment（显式旗标优先，其次首个可绑定结果）。
   303	    sha = (args.sha or "").strip().lower() or None
   304	    env = (args.env or "").strip() or None
   305	    for doc in docs:
   306	        if not isinstance(doc, Mapping):
   307	            continue
   308	        identity = doc.get("identity")
   309	        if not isinstance(identity, Mapping):
   310	            continue
   311	        if sha is None:
   312	            raw = identity.get("commit_sha")
   313	            if isinstance(raw, str) and raw.strip():
   314	                sha = raw.strip().lower()
   315	        if env is None:
   316	            raw = identity.get("environment")
   317	            if isinstance(raw, str) and raw.strip():
   318	                env = raw.strip()
   319	        if sha is not None and env is not None:
   320	            break
   321	    if sha is None or env is None:
   322	        missing = "target_sha" if sha is None else "environment"
   323	        return _gate_error_json(
   324	            f"无法绑定 {missing}：--sha/--env 未给且没有任何结果携带可用的 "
   325	            f"identity.{missing}（gate 证据必须绑定 SHA+环境）",
   326	            inputs={"paths": len(paths), "docs": len(docs),
   327	                    "load_errors": len(load_errors), "schema_invalid": 0},
   328	            load_errors=load_errors)
   329	
   330	    # 4. 逐个严格校验（schema 可执行等价物）＋收集默认 required 集。
   331	    strict_violations: List[Dict[str, str]] = []
   332	    default_required: Set[str] = set()
   333	    for doc in docs:
   334	        try:
   335	            validate_station_result(doc)
   336	        except StationResultValidationError as exc:
   337	            strict_violations.append({"error": str(exc)})
   338	        if isinstance(doc, Mapping):
   339	            sid = doc.get("station_id")
   340	            if isinstance(sid, int) and not isinstance(sid, bool):
   341	                default_required.add(str(sid))
   342	            elif isinstance(doc.get("station"), str) and doc.get("station").strip():
   343	                default_required.add(doc["station"].strip())
   344	
   345	    if args.required is not None:
   346	        required: Set[str] = set()
   347	        for token in args.required.split(","):
   348	            token = token.strip()
   349	            if not token:
   350	                continue
   351	            if not (token.isdigit() and 0 <= int(token) <= 7):
   352	                return _gate_error_json(
   353	                    f"--required 站 id 必须是 0-7 的整数: {token!r}",
   354	                    inputs={"paths": len(paths), "docs": len(docs),
   355	                            "load_errors": len(load_errors),
   356	                            "schema_invalid": len(strict_violations)})
   357	            required.add(token)
   358	        if not required:
   359	            return _gate_error_json(
   360	                "--required 为空：显式必报站不能是空集（真空绿拒绝）",
   361	                inputs={"paths": len(paths), "docs": len(docs),
   362	                        "load_errors": len(load_errors),
   363	                        "schema_invalid": len(strict_violations)})
   364	    else:
   365	        required = default_required
   366	
   367	    # 5. 聚合（GateAggregator 宽松读+严格出；C1-C5 全在 aggregate 内裁决）。
   368	    agg = GateAggregator(required_stations=required, target_sha=sha,
   369	                         environment=env)
   370	    for doc in docs:
   371	        agg.add(doc)
   372	    result = agg.aggregate()
   373	
   374	    # 6. 结构校验失败强制 BLOCKED（绝不静默跳过残缺结果——跳过即分母缩水）。
   375	    if strict_violations:
   376	        if result.get("aggregate_outcome") != BLOCKED:
   377	            result["aggregate_outcome"] = BLOCKED
   378	            result["policy_verdict"] = BLOCKED
   379	            result["technical_eligible"] = False
   380	        first = strict_violations[0]["error"]
   381	        reason = (f"station-result-v2 schema violation x{len(strict_violations)} "
   382	                  f"[first={first[:160]}]")
   383	        result["reasons"] = [reason] + list(result.get("reasons", []))
   384	        result["reason"] = reason
   385	
   386	    result["inputs"] = {
   387	        "paths": len(paths),
   388	        "docs": len(docs),
   389	        "load_errors": len(load_errors),
   390	        "schema_invalid": len(strict_violations),
   391	    }
   392	    if load_errors:
   393	        result["input_errors"] = load_errors
   394	
   395	    print(json.dumps(result, ensure_ascii=False))
   396	    return _gate_exit_code(result["aggregate_outcome"])

    82	def _sha256_file(path: str) -> str:
    83	    digest = hashlib.sha256()
    84	    with open(path, "rb") as fh:
    85	        for chunk in iter(lambda: fh.read(_CHUNK), b""):

   108	    def __init__(self, releases_root: str, current_link: str,

   115	    # 构建：不可变 release 目录 + manifest
   116	    # ------------------------------------------------------------------ #
   117	    def build(self, source_dir: str) -> Dict[str, Any]:
   118	        """将 source_dir 构建为不可变 release 目录（全文件 SHA256+mode manifest）。
   119	
   120	        D1：staging 全新构建 + rename 原子发布；绝不向既有 release 写入/覆盖。
   121	        同秒同内容重复构建 -> 幂等复用已存在且完好的 release（reused=True）；
   122	        release_id 冲突但内容 divergent/损坏 -> BuildError。
   123	        P0-7 二次修：mode 策略按「源文件 exec 位」判定——源文件含任一 exec 位
   124	        （0755 无扩展可执行文件同理）发布后保持可执行（0555）；否则 0444
   125	        （0644 带 shebang 的 .py 不强加 exec）。manifest 逐文件记录 mode，
   126	        verify() 对账 mode 漂移。
   127	        """
   128	        source = os.path.abspath(source_dir)
   129	        if not os.path.isdir(source):
   130	            raise BuildError(f"source_dir not found: {source}")
   131	        os.makedirs(self.releases_root, exist_ok=True)
   132	
   133	        rel_files = _collect_files(source)
   134	        staging = tempfile.mkdtemp(prefix=".staging-", dir=self.releases_root)
   135	        try:
   136	            entries: List[Dict[str, Any]] = []
   137	            digest = hashlib.sha256()
   138	            for rel in rel_files:
   139	                src = os.path.join(source, rel)
   140	                dst = os.path.join(staging, rel)
   141	                os.makedirs(os.path.dirname(dst), exist_ok=True)
   142	                file_hash = self._copy_stream_hash(src, dst)  # 逐文件流式复制，非 cp -R
   143	                # P0-7 修：mode 策略——源文件 exec 位决定发布后可执行性
   144	                src_mode = os.stat(src).st_mode & 0o777
   145	                exec_bit = bool(src_mode & 0o111)
   146	                release_mode = 0o555 if exec_bit else 0o444
   147	                os.chmod(dst, release_mode)
   148	                entries.append({
   149	                    "path": rel,
   150	                    "sha256": file_hash,
   151	                    "size": os.path.getsize(dst),
   152	                    "mode": "%04o" % release_mode,
   153	                })
   154	                digest.update(rel.encode("utf-8") + b"\0" + file_hash.encode("ascii") + b"\0")
   155	            content_digest = digest.hexdigest()
   156	
   157	            release_id = f"{time.strftime('%Y%m%dT%H%M%SZ', time.gmtime())}-{content_digest[:12]}"
   158	            final_dir = os.path.join(self.releases_root, release_id)
   159	
   160	            manifest = {
   161	                "release_id": release_id,
   162	                "created_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
   163	                "source_dir": source,
   164	                "content_digest": content_digest,
   165	                "files": sorted(entries, key=lambda e: e["path"]),
   166	            }
   167	            manifest_path = os.path.join(staging, MANIFEST_NAME)
   168	            with open(manifest_path, "w", encoding="utf-8") as fh:
   169	                json.dump(manifest, fh, ensure_ascii=False, indent=2, sort_keys=True)
   170	                fh.write("\n")
   171	            os.chmod(manifest_path, 0o444)
   172	
   173	            if os.path.exists(final_dir):
   174	                # 幂等复用：仅当既有 release 完好且内容摘要一致
   175	                if self._reusable(final_dir, content_digest):
   176	                    shutil.rmtree(staging, ignore_errors=True)
   177	                    with open(os.path.join(final_dir, MANIFEST_NAME), encoding="utf-8") as fh:
   178	                        existing = json.load(fh)
   179	                    return {
   180	                        "release_dir": final_dir,
   181	                        "manifest_path": os.path.join(final_dir, MANIFEST_NAME),
   182	                        "release_id": release_id,
   183	                        "content_digest": content_digest,
   184	                        "files": len(existing.get("files", [])),
   185	                        "reused": True,
   186	                    }
   187	                raise BuildError(
   188	                    f"release_id collision with divergent/broken existing release: {final_dir}")
   189	
   190	            os.rename(staging, final_dir)  # D3：同目录 rename，原子发布
   191	        except BaseException:
   192	            shutil.rmtree(staging, ignore_errors=True)
   193	            raise
   194	
   195	        return {
   196	            "release_dir": final_dir,
   197	            "manifest_path": os.path.join(final_dir, MANIFEST_NAME),
   198	            "release_id": release_id,
   199	            "content_digest": content_digest,
   200	            "files": len(entries),
   201	            "reused": False,
   202	        }

   204	    @staticmethod
   205	    def _copy_stream_hash(src: str, dst: str) -> str:
   206	        """流式复制并计算 SHA256（边写边哈希，单次读盘）。"""
   207	        digest = hashlib.sha256()
   208	        with open(src, "rb") as fsrc, open(dst, "wb") as fdst:
   209	            for chunk in iter(lambda: fsrc.read(_CHUNK), b""):
   210	                digest.update(chunk)
   211	                fdst.write(chunk)
   212	        return digest.hexdigest()

   214	    def _reusable(self, final_dir: str, content_digest: str) -> bool:
   215	        try:
   216	            with open(os.path.join(final_dir, MANIFEST_NAME), encoding="utf-8") as fh:
   217	                existing = json.load(fh)

   219	                return False

   228	    def _resolve_link_value(link_dir: str, value: str) -> str:
   229	        """相对链接值一律相对 link 所在目录解析（绝不相对进程 cwd）。
   230	
   231	        绝对值原样 normpath 返回。P0-7 修：previous/current 相对链接曾按

   238	    def verify(self, release_dir: str, strict: bool = True) -> Dict[str, Any]:
   239	        """manifest 对账：缺件/多件/哈希漂移/mode 漂移任一非空即失败（D4）。
   240	
   241	        strict=True（默认）失败抛 VerifyError（携带完整报告）；
   242	        strict=False 返回含 ``ok`` 布尔的报告，供调用方自行裁决。
   243	        P0-7 二次修：manifest 逐文件记录 mode；发布后 chmod 篡改（如 0444→
   244	        0755）在 verify 即失败（mode_drifted），不再只校验内容哈希。
   245	        """
   246	        release = os.path.abspath(release_dir)
   247	        manifest_path = os.path.join(release, MANIFEST_NAME)
   248	        if not os.path.isfile(manifest_path):
   249	            report = {
   250	                "ok": False, "release_dir": release, "release_id": None,
   251	                "missing": [MANIFEST_NAME], "extra": [], "drifted": [],
   252	                "mode_drifted": [], "checked": 0,
   253	            }
   254	            if strict:
   255	                raise VerifyError(report)
   256	            return report
   257	
   258	        with open(manifest_path, encoding="utf-8") as fh:
   259	            manifest = json.load(fh)
   260	        entries = {e["path"]: e for e in manifest.get("files", [])}
   261	        expected = {p: e["sha256"] for p, e in entries.items()}
   262	        actual = set(_collect_files(release)) - {MANIFEST_NAME}  # manifest 自身不在清单内
   263	        expected_set = set(expected)
   264	
   265	        missing = sorted(expected_set - actual)
   266	        extra = sorted(actual - expected_set)
   267	        drifted = sorted(
   268	            p for p in expected_set & actual
   269	            if _sha256_file(os.path.join(release, p)) != expected[p]
   270	        )
   271	        # P0-7 修：mode 对账——manifest 记录了 mode 的文件，实际权限位必须一致
   272	        mode_drifted = sorted(
   273	            p for p in expected_set & actual
   274	            if entries[p].get("mode") is not None
   275	            and _stat_mode(os.path.join(release, p)) != int(entries[p]["mode"], 8)
   276	        )
   277	        ok = not (missing or extra or drifted or mode_drifted)
   278	        report = {
   279	            "ok": ok,
   280	            "release_dir": release,
   281	            "release_id": manifest.get("release_id"),
   282	            "missing": missing,
   283	            "extra": extra,
   284	            "drifted": drifted,
   285	            "mode_drifted": mode_drifted,
   286	            "checked": len(expected_set & actual),
   287	        }
   288	        if strict and not ok:
   289	            raise VerifyError(report)
   290	        return report

   295	    def switch(self, current_symlink: str, target: str) -> Dict[str, Any]:
   296	        """原子切换 current symlink -> target（D3：临时链接 + os.replace）。"""
   297	        link = os.path.abspath(current_symlink)
   298	        target_abs = os.path.abspath(target)
   299	        if not os.path.isdir(target_abs):
   300	            raise SwitchError(f"switch target not found: {target_abs}")
   301	
   302	        link_dir = os.path.dirname(link) or "."
   303	        os.makedirs(link_dir, exist_ok=True)
   304	        if os.path.lexists(link) and not os.path.islink(link):
   305	            raise SwitchError(f"refuse to replace non-symlink path: {link}")
   306	
   307	        previous = os.readlink(link) if os.path.islink(link) else None
   308	        link_value = os.path.relpath(target_abs, link_dir)  # 相对链接，目录整体迁移不失效
   309	        tmp_link = os.path.join(
   310	            link_dir, f".{os.path.basename(link)}.switch-{os.getpid()}")
   311	        if os.path.lexists(tmp_link):
   312	            os.remove(tmp_link)
   313	        os.symlink(link_value, tmp_link)
   314	        try:
   315	            os.replace(tmp_link, link)  # D3：rename 原子性，瞬时生效
   316	        except OSError as exc:
   317	            if os.path.lexists(tmp_link):
   318	                os.remove(tmp_link)
   319	            raise SwitchError(f"atomic symlink switch failed: {link} -> {target_abs}: {exc}") from exc
   320	
   321	        return {
   322	            "current_symlink": link,
   323	            "previous": previous,
   324	            "previous_abs": (
   325	                self._resolve_link_value(link_dir, previous) if previous else None),
   326	            "target": target_abs,
   327	            "link_value": link_value,
   328	        }

   330	    def rollback(self, current: str, previous: str) -> Dict[str, Any]:
   331	        """切回兼容版：先对 previous release 做 manifest 完好性对账，再原子切回。
   332	
   333	        previous 已损坏（缺件/漂移）时拒绝回切并抛错——绝不把流量切到坏包上。
   334	        P0-7 修：previous 允许是相对链接值——一律相对 current 链接所在目录解析
   335	        （不相对进程 cwd）。
   336	        """
   337	        link = os.path.abspath(current)
   338	        link_dir = os.path.dirname(link) or "."
   339	        prev = self._resolve_link_value(link_dir, previous)
   340	        if not os.path.isdir(prev):
   341	            raise SwitchError(f"rollback target missing: {prev}")
   342	        verify_report = self.verify(prev)
   343	        result = self.switch(link, prev)
   344	        result.update({"rolled_back": True, "verify": verify_report})
   345	        return result

   350	    def deploy(self, source_dir: str) -> Dict[str, Any]:
   351	        """完整部署：构建 -> 对账 -> 原子切换 -> 安装自检。
   352	
   353	        D2：自检失败 = 部署失败——自动回滚上一版后抛 DeployError。
   354	        P0-7 修：首次部署无上一版可回滚时，切回空/删除 current link（不留在失败版上）。
   355	        P0-7 二次修：previous 相对链接值一律相对 current 链接所在目录解析
   356	        （原按进程 cwd 解析 → 误判不存在 → 跳过回滚 → current 留在失败版）；
   357	        回滚后对回滚目标再跑一次 self-check 复验。
   358	        """
   359	        build_result = self.build(source_dir)
   360	        verify_report = self.verify(build_result["release_dir"])  # 切换前对账
   361	
   362	        # P0-7 修：切换前记录是否存在 current link（首次部署判定）
   363	        _had_current = os.path.islink(self.current_link) or os.path.exists(self.current_link)
   364	        _prev_target_raw = None
   365	        _prev_target_abs = None
   366	        if _had_current and os.path.islink(self.current_link):
   367	            _prev_target_raw = os.readlink(self.current_link)
   368	            # P0-7 修：相对链接值按 link 所在目录解析（不相对进程 cwd）
   369	            _prev_target_abs = self._resolve_link_value(
   370	                os.path.dirname(self.current_link) or ".", _prev_target_raw)
   371	
   372	        switch_report = self.switch(self.current_link, build_result["release_dir"])
   373	
   374	        try:
   375	            if self.install_self_check is not None:
   376	                check = self.install_self_check(build_result["release_dir"])
   377	                if not check:
   378	                    raise SelfCheckError(f"install self-check returned falsy: {check!r}")
   379	        except Exception as exc:
   380	            rollback_note = "no previous release to roll back to"
   381	            if _prev_target_abs and os.path.isdir(_prev_target_abs):
   382	                try:
   383	                    self.rollback(self.current_link, _prev_target_abs)
   384	                    rollback_note = f"rolled back to {_prev_target_abs}"
   385	                    # P0-7 修：回滚后对回滚目标再验一次 self-check
   386	                    if self.install_self_check is not None:
   387	                        try:
   388	                            recheck = self.install_self_check(_prev_target_abs)
   389	                            rollback_note += (
   390	                                "; rollback self-check re-verified"
   391	                                if recheck else
   392	                                "; WARNING: rollback target also fails self-check"
   393	                            )
   394	                        except Exception as sc_exc:
   395	                            rollback_note += (
   396	                                f"; WARNING: rollback target self-check error: {sc_exc!r}")
   397	                except DeployError as rb_exc:
   398	                    rollback_note = (f"ROLLBACK FAILED ({rb_exc}); "
   399	                                     f"current link remains on failed release "
   400	                                     f"{build_result['release_dir']}")
   401	            elif not _had_current:
   402	                # P0-7 修：首次部署失败——删除 current link（不留在失败版上）
   403	                try:
   404	                    os.unlink(self.current_link)
   405	                    rollback_note = "first deploy: current link removed (no failed release active)"
   406	                except OSError:
   407	                    rollback_note = f"first deploy: failed to remove current link, stays on failed release"
   408	            raise DeployError(
   409	                f"deploy failed at install self-check: {exc!r}; {rollback_note}"
   410	            ) from exc
   411	
   412	        return {
   413	            "deployed": True,
   414	            "release_id": build_result["release_id"],
   415	            "release_dir": build_result["release_dir"],
   416	            "build": build_result,
   417	            "verify": verify_report,
   418	            "switch": switch_report,
   419	            "self_check": "passed",
